\section{Branching with variable zeroing}
\label{sec:variable-zeroing}

\subsection{Branching with variable zeroing}
\label{subsec:zeroing-branching}

% Motivation for splitting
Lets say that the solver doesn't terminate within reasonable amount of time for an hitting set given by Algorithm~\ref{alg:Branching}.

% What is splitting
In this case, we can further simplify the system by branching further.
In each branch assume some side condition holds, and letting Algorithm~\ref{alg:Branching} run on the simplified system.
This splitting can be done recursively until the solver terminates or the computation budget is exhausted.

% Good properties for conditions on the set of conditions
For completeness, the union of the side conditions must cover all possibilities.
A set of conditions is a good choice if it is helpful for your solver, and works well with algorithm~\ref{alg:Refine}.

% What is the condition we use, and why?
For this work the branching conditions is whether a variable is zero or nonzero.
This simplification choice is used in other works~\cite{}. % Add citation.
in this work we considered physical systems and determined through past experience that we expect a variable to be always zero, or always nonzero.

% Pseudocode for the method with variable zeroing
\begin{algorithm}[H]
\small
\caption{Recursive branching with variable zeroing}
\label{alg:ZeroingBranching}
\begin{algorithmic}[1]
    \Require A polynomial system $P$ and a timeout $\tau$
    \Function{ZeroingBranching}{$P,\tau$}
        \State $X \gets \Call{Variables}{P}$
        \State \Return $\Call{SolveByZeroing}{P,X,\emptyset,\emptyset,\tau,\emptyset}$
    \EndFunction
    \Function{SolveByZeroing}{$Q,X,Z,N,\tau,C$}
        \State $(S,T) \gets \Call{BranchingWithTimeout}{Q,\tau,C}$
        \State $U \gets \emptyset$ \Comment{Unresolved branches}

        \ForAll{$R \in T$}
            \If{$X=\emptyset$}
                \State $U \gets U\cup\{(R,Z,N)\}$
            \Else
                \State $x \gets \Call{ChooseVariable}{R,X}$
                \State $R_0 \gets \Call{Substitute}{R,x,0}$
                \State $R_1 \gets \Call{UpdateSaturation}{R,N\cup\{x\}}$
                \State $(S_0,U_0,C) \gets \Call{SolveByZeroing}{R_0,X\setminus\{x\},Z\cup\{x\},N,\tau,C}$
                \State $(S_1,U_1,C) \gets \Call{SolveByZeroing}{R_1,X\setminus\{x\},Z,N\cup\{x\},\tau,C}$
                \State $S \gets S\cup S_0\cup S_1$
                \State $U \gets U\cup U_0\cup U_1$
            \EndIf
        \EndFor
        \State \Return $(S,U,C)$
    \EndFunction
\end{algorithmic}
\end{algorithm}

For Algorithm~\ref{alg:ZeroingBranching} we use a global set $C$ of learned UNSAT cores.

% How the constraints are encoded into the system.
Let $N$ be the variables declared nonzero on the current path.
Subsection~\ref{subsec:nonzero-encoding} describes two ways to encode these
conditions as polynomial equations.

Solutions of the nonzero branch are projected onto the original variables, so
the saturation variable does not appear in the output. The zero and nonzero
decisions remain part of the solution-set representation.

\subsection{Choice of variable for splitting}
\label{subsec:zeroing-variable-choice}

The choice of variable for splitting affects the efficiency of the algorithm.
Every split creates at least two new hitting sets to check.
If a variable barely occurs in the current hitting set, both recursive calls may repeat most of the same work.
A variable that occurs in many generators is a more promising choice. Setting
it to zero removes all terms containing it at once. In the nonzero branch, it
is appended to the product in Equation~\eqref{eq:zeroing-saturation}.

For each candidate variable $x$ that is still available on the current branch, define
\begin{align*}
    g_Q(x) &= \#\{q\in Q : x\text{ occurs in }q\},\\
    t_Q(x) &= \#\{(q,m) : q\in Q,\ m\text{ is a term of }q,\ \deg_x(m) > 0\},\\
    e_Q(x) &= \sum_{q\in Q}\ \sum_{\substack{m\text{ a term of }q,\\ \deg_x(m) > 0}} \deg_x(m).
\end{align*}
Its score is the triple
\[
    h_Q(x)=\bigl(g_Q(x),t_Q(x),e_Q(x)\bigr),
\]
and scores are compared lexicographically.

\subsection{Branch contraction}
\label{subsec:zeroing-branch-contraction}

Let $\mathcal H_0$ and $\mathcal H_1$ be the minimal hitting sets for $x=0$ and $x\neq 0$.
Suppose, up to exchanging the two children, that
\[
    \mathcal H_0=\emptyset
    \qquad\text{and}\qquad
    \mathcal H_1=\{H\}.
\]
The first child is inconsistent, and we have
\[
    V_Y(Q)=V_Y(H).
\]
There is no need to run the branch solver on $H$ before choosing the next
splitting variable.

\subsection{Theoretical analysis of variable zeroing}
\label{subsec:zeroing-theoretical-analysis}

We prove the correctness and termination of Algorithm~\ref{alg:ZeroingBranching}.

Let $V_Y(Q;Z,N)$ denote the projection onto $Y$ of the solutions of $Q$ satisfying $z=0$ for $z\in Z$ and $y\neq0$ for $y\in N$.
We also denote $Q_0=Q|_{x=0}$ and $Q_1=\Call{UpdateSaturation}{Q,N\cup\{x\}}$.

\begin{theorem}[Correctness of a zeroing split]
\label{lem:zeroing-split}
Let there be $Z,N\subseteq Y$ such that $Z \cap N = \emptyset$, and let $x\in Y\setminus(Z\cup N)$.
Then
\begin{align*}
    V_Y(Q;Z,N)=V_Y(Q_0;Z\cup\{x\},N) \cup V_Y(Q_1;Z,N\cup\{x\}).
\end{align*}
\end{theorem}

\begin{proof}
Take a point in $V_Y(Q;Z,N)$. Its $x$-coordinate is either zero or
nonzero. In the first case, substituting $x=0$ gives a solution of $Q_0$
satisfying $Z\cup\{x\}$ and $N$. In the second case, the point satisfies the
nonzero conditions $N\cup\{x\}$, which are encoded by $Q_1$. Thus it belongs
to the corresponding set on the right.

Conversely, a point in either set on the right satisfies $Q$, together with
the conditions in $Z$ and $N$. Therefore it belongs to $V_Y(Q;Z,N)$.
\end{proof}

\begin{theorem}[Correctness of variable zeroing]
\label{thm:zeroing-correctness}
If Algorithm~\ref{alg:ZeroingBranching} returns $(S,U,C)$ on input $P$, then

\begin{align*}
    V(P)=\operatorname{Sol}(S) \cup\bigcup_{(R,Z_R,N_R)\in U}V_Y(R;Z_R,N_R).
\end{align*}
\end{theorem}

\begin{proof}
We prove the claim for every call to \Call{SolveByZeroing} by induction on
the remaining recursion tree. For a call with input $(Q,X,Z,N,\tau,C)$,
\Call{BranchingWithTimeout} partitions the constrained zero set into the
solutions it finds and the unresolved hitting sets:
\[
    V_Y(Q;Z,N)=\operatorname{Sol}(S)
    \cup\bigcup_{R\in T}V_Y(R;Z,N).
\]
This follows from Theorem~\ref{thm:branching-correctness}; a timeout leaves
the corresponding hitting set in $T$, and pruning by a certified UNSAT core
removes only an empty zero set.

If $X=\emptyset$, the algorithm adds each $R\in T$ to $U$ with the current
conditions, so the required equality follows. Otherwise, for every $R\in T$
the algorithm chooses $x\in X$. By Theorem~\ref{lem:zeroing-split},
\[
    V_Y(R;Z,N)=V_Y(R_0;Z\cup\{x\},N)
    \cup V_Y(R_1;Z,N\cup\{x\}).
\]
The induction hypothesis applies to both recursive calls. Their solved and
unresolved sets therefore replace $V_Y(R;Z,N)$ without changing the union.
Taking the union over all $R\in T$ proves the claim for the current call.

The initial call has $Q=P$ and $Z=N=\emptyset$, so its constrained zero set
is $V(P)$. This gives the stated equality.
\end{proof}

The following corollary is a direct consequence of Theorem~\ref{thm:zeroing-correctness}.
It states that if all branches are resolved, Algorithm~\ref{alg:ZeroingBranching} returns a complete solution set.
\begin{corollary}[Completeness of variable zeroing]
\label{cor:zeroing-completeness}
If $U = \emptyset$ in the output of Algorithm~\ref{alg:ZeroingBranching}, then
\[
    \operatorname{Sol}(S)=V(P).
\]
\end{corollary}

Lastly we prove that Algorithm~\ref{alg:ZeroingBranching} terminates.
\begin{theorem}[Termination of variable zeroing]
\label{thm:zeroing-termination}
Algorithm~\ref{alg:ZeroingBranching} terminates.
\end{theorem}

\begin{proof}
We prove that \Call{SolveByZeroing} terminates by induction on $|X|$. First,
every call to \Call{BranchingWithTimeout} terminates. Indeed,
Theorem~\ref{thm:refine-termination} shows that Algorithm~\ref{alg:Refine} terminates, and returns finitely many hitting sets.
On there minimal hitting sets, the solver either returns or is stopped after the timeout $\tau$. Thus the call returns finite sets $S$ and $T$.

If $X=\emptyset$, the algorithm loops over the finite set $T$ and returns.
Suppose now that $|X|>0$. For each $R\in T$, the algorithm makes two recursive
calls, both with $X\setminus\{x\}$. These calls terminate by the induction
hypothesis. Since $T$ is finite, the current call also terminates.

The initial set $\Call{Variables}{P}$ is finite, so the initial call to
\Call{SolveByZeroing} terminates.
Therefore Algorithm~\ref{alg:ZeroingBranching} terminates.
\end{proof}
