\section{Ideal-preserving factor branching}
\label{sec:ideal-branching}

% What Algorithm Branching loses: the exponents, and with them the ideal
Algorithm~\ref{alg:Branching} discards the exponents $e_{q,f}$ and hence only
describes $V(P)$ and $\sqrt{\langle P\rangle}$. But ideal membership,
multiplicities, and embedded components depend on $I=\langle P\rangle$ itself.
This section adapts factor branching so that it preserves the ideal.

\subsection{Ideal reduction}
\label{subsec:sat-reduction}

% Notation shared by the section
For a finite system $Q$ and $h\in C(Q)$, let $a_h=\max\{e_{q,h}:q\in Q,\ h\in C_q\}$ be
the largest exponent of $h$.

% The rule: a factor chosen by every SAT branch lies in the ideal
Suppose that a factor $h$ lies in every SAT minimal hitting set of $Q$. Then
$h$ vanishes on all of $V(Q)$, and the Nullstellensatz puts some power of $h$
in $\langle Q\rangle$. Lemma~\ref{lem:sat-branches} shows that the power
$h^{a_h}$ suffices, so the factorization bounds the exponent without a
Gr\"obner basis of $\langle Q\rangle$.

% Worked example: the rule finds a member that interreduction misses
For example, $Q=\{xy,\,xz,\,yz-1\}$ is interreduced for every monomial order.
Its factor sets $\{x,y\}$, $\{x,z\}$, and $\{yz-1\}$ have the minimal hitting
sets $\{x,yz-1\}$ and $\{y,z,yz-1\}$. The second is UNSAT, since $y=z=0$
contradicts $yz-1=0$. Hence $x$ and $yz-1$ are in every SAT minimal hitting
set, $a_x = 1$ and $a_{yz-1} = 1$, and indeed $x^1,
(yz-1)^1\in\langle Q\rangle$.

% The reduction loop
Algorithm~\ref{alg:IdealReduce} adds each $h^{a_h}$ to the generators, drops
the generators they divide, and repeats on the new system. We say
that $\mathcal B$ certifies $V(H)=\emptyset$ when Algorithm~\ref{alg:Branching}
with $\mathcal B$ certifies every terminal branch of $H$ as UNSAT. If no
minimal hitting set is SAT, the intersection is $C(Q)$, and
Lemma~\ref{lem:sat-branches} still applies. On the example, the first round
replaces $xy$ and $xz$ by $x$, giving $\{x,yz-1\}$.

\begin{algorithm}[htbp]
  \caption{Ideal reduction}
  \label{alg:IdealReduce}
  \begin{algorithmic}[1]
    \Require A finite polynomial system $P$ and a branch solver $\mathcal B$
    \Ensure A finite generating set $Q$ of $\langle P\rangle$
    \State $Q\gets\Call{Interreduce}{P}$
    \Loop
    \State $\mathcal H_{\mathrm{sat}}\gets\{H\in\mathcal H(Q):\mathcal B
    \text{ does not certify }V(H)=\emptyset\}$
    \State $\Delta\gets\{h^{a_h}:h\in\bigcap\mathcal
    H_{\mathrm{sat}}\}\setminus Q$
    \If{$\Delta=\emptyset$}
    \State \Return $Q$
    \EndIf
    \State $Q\gets\Call{Interreduce}{\Delta\cup\{q\in Q:\text{no
          element of $\Delta$
    divides $q$}\}}$
    \EndLoop
  \end{algorithmic}
\end{algorithm}

% What the reduced system is good for
The output generates $\langle P\rangle$, with polynomials replaced by a
factor to the largest exponent that appears in the system, yielding a simpler system.

% Optimization: only the intersection is needed, so most solver calls
% can be skipped
Each round needs only the intersection of the SAT minimal hitting sets, so an
implementation need not classify every minimal hitting set. It keeps the
factors $F$ that lie in every SAT hitting set checked so far. Starting from
$F=C(Q)$, it skips every hitting set that contains $F$, intersects $F$ with
every other hitting set that is SAT, and stops once $F$ is empty.
Each skipped hitting set contains $F$, so the final $F$ is
$\bigcap\mathcal H_{\mathrm{sat}}$. Two disjoint SAT hitting sets already
empty $F$, so a round that finds nothing often costs only a few solver calls.

\subsection{Primary factor branching}
\label{subsec:primary-decomposition}

% What the earlier variants leave open
Algorithm~\ref{alg:Branching} with $\mathcal B_{\min}$ describes only
$\sqrt I$. A minimal primary decomposition of $I$ also records the
multiplicity of each component and embedded components.
Algorithm~\ref{alg:PrimaryBranching} computes one with an algorithm
$\mathcal B_{\mathrm{prim}}$ that returns a minimal primary decomposition of
its input. Its other steps make $\mathcal B_{\mathrm{prim}}$ cheaper, skip it,
or label its output.

% Step 1: a smaller system for the same ideal
The algorithm first replaces $P$ by the output $Q$ of
Algorithm~\ref{alg:IdealReduce}. By Theorem~\ref{thm:ideal-reduce},
$\langle Q\rangle=I$, so every later step works on a smaller description of $I$.

% Step 2: the minimal primes
% The reduced system gives a second route to the same primes
% Optimizations: factor branching for the minimal primes
It then computes the minimal primes $\mathfrak p_1,\ldots,\mathfrak p_r$ of
$I$, which give the isolated components in the optional lines and mark the
components of $\mathcal B_{\mathrm{prim}}$. Any algorithm $\mathcal B_{\min}$
that returns the minimal primes of its input works here, and we use
Theorem~\ref{thm:prime-branching} in the experiments
(Section~\ref{sec:experiments}). It finds the minimal primes without
decomposing $I$ itself, and it can skip the minimal hitting sets of $Q$ that
the last round of Algorithm~\ref{alg:IdealReduce} certified as UNSAT.

% Optimization: isolated components by separators, without the general solver
Optionally, the minimal primes give the isolated components
$\mathfrak q_1,\ldots,\mathfrak q_r$ without $\mathcal B_{\mathrm{prim}}$.
They intersect to $I$ exactly when $I$ has no embedded components, and the
algorithm then returns them, so these lines pay off only on such ideals. Each
$\mathfrak q_i$ comes from a \emph{separator} of $\mathfrak p_i$, a
polynomial $w_i\in\bigcap_{j\neq i}\mathfrak p_j\setminus\mathfrak p_i$,
which exists because $\mathfrak p_i$ contains no other minimal prime and hence,
by Subsection~\ref{subsec:decompositions}, not their intersection. Saturating
by $w_i$ removes every component whose radical does not contain
$\mathfrak p_i$~\cite{shimoyama1996localization}, so $\mathfrak q_i$ is the
top-dimensional part of $I:w_i^\infty$, the intersection of its primary
components of maximal dimension~\cite{eisenbud1992direct}
(Lemma~\ref{lem:isolated-separator}). A Gr\"obner basis of $I$ decides whether
$\mathfrak q_1\cap\cdots\cap\mathfrak q_r=I$.

% Step 3: one general primary decomposition of the reduced system
Otherwise, $\mathcal B_{\mathrm{prim}}$ decomposes $\langle Q\rangle$, which
is $I$.

% Step 4: the minimal primes sort the components
A component $\mathfrak q$ is isolated if $\sqrt{\mathfrak q}$ is one of the
$\mathfrak p_i$ and embedded otherwise. The radical of an embedded component
strictly contains at least one $\mathfrak p_i$, so the variety of the component
lies inside an irreducible component of $V(I)$.

% Worked example: an embedded component
For example, $I=\langle x^2,xy\rangle$ has the single minimal prime
$\langle x\rangle$, which needs no separator. The top-dimensional part of $I$
is $\langle x\rangle\neq I$, so $\mathcal B_{\mathrm{prim}}$ runs and returns
$I=\langle x\rangle\cap\langle x^2,y\rangle$. The component $\langle x\rangle$
is isolated, and $\langle x^2,y\rangle$ is embedded, since its radical
$\langle x,y\rangle$ strictly contains $\langle x\rangle$.

\begin{algorithm}[htbp]
  \caption{Primary factor branching}
  \label{alg:PrimaryBranching}
  \begin{algorithmic}[1]
    \Require A finite polynomial system $P$, branch solvers $\mathcal B$ and
    $\mathcal B_{\min}$, and a minimal primary decomposition algorithm
    $\mathcal B_{\mathrm{prim}}$
    \Ensure A minimal primary decomposition of $\langle P\rangle$, each
    component marked isolated or embedded
    \State $Q\gets\Call{IdealReduce}{P,\mathcal B}$
    \Comment{Algorithm~\ref{alg:IdealReduce}, $\langle
    Q\rangle=\langle P\rangle$}
    \State $\{\mathfrak p_1,\ldots,\mathfrak p_r\}\gets\mathcal B_{\min}(Q)$
    \par\vskip3pt\hrule\vskip3pt
    \ForAll{$i\in\{1,\ldots,r\}$} \Comment{Optional}
    \State $w_i\gets$ a separator of $\mathfrak p_i$
    \State $\mathfrak q_i\gets$ the top-dimensional part of
    $\langle Q\rangle:w_i^\infty$
    \EndFor
    \If{$\mathfrak q_1\cap\cdots\cap\mathfrak q_r=\langle Q\rangle$}
    \State \Return $\{\mathfrak q_1,\ldots,\mathfrak q_r\}$, all marked
    isolated
    \EndIf
    \par\vskip3pt\hrule\vskip3pt
    \State $\mathcal D\gets\mathcal B_{\mathrm{prim}}(Q)$
    \ForAll{$\mathfrak q\in\mathcal D$}
    \State mark $\mathfrak q$ isolated if
    $\sqrt{\mathfrak q}\in\{\mathfrak p_1,\ldots,\mathfrak p_r\}$,
    and embedded otherwise
    \EndFor
    \State \Return $\mathcal D$
  \end{algorithmic}
\end{algorithm}

% Optimization: a separator from one saturation
A separator also needs no intersection of primes. The saturation
$I:\mathfrak p_i^\infty=\{g\in R:\mathfrak p_i^mg\subseteq I\text{
for some }m\geq0\}$
keeps exactly the components whose radicals do not contain $\mathfrak p_i$,
so it lies in every other minimal prime but not in $\mathfrak p_i$, and every
element of its Gr\"obner basis outside $\mathfrak p_i$ is a separator.

% Optimization: zero-dimensional ideals need no separators and have
% no embedded components
A zero-dimensional $I$ needs neither separators nor the intersection test:
its primes are maximal, so it has no embedded components. Its components are
pairwise comaximal, so $\dim_K R/I$ is the sum of the $\dim_K R/\mathfrak q_i$,
each at least $\dim_K R/\mathfrak p_i$ with equality only if
$\mathfrak q_i=\mathfrak p_i$. If $\dim_KR/I=\sum_i\dim_KR/\mathfrak p_i$,
every component is prime. Otherwise $\mathfrak q_i=I+\mathfrak p_i^m$ for the
first $m$ at which $\dim_K R/(I+\mathfrak p_i^m)$ stops growing, by Nakayama's
lemma~\cite[Prop.~2.6]{atiyah1969introduction}.

% Implementation: comparing a radical with the minimal primes
The mark needs no equality test of ideals. A prime that strictly contains
$\mathfrak p_i$ has smaller dimension
(Subsection~\ref{subsec:minimal-prime-decomposition}), so $\mathfrak q$ is
isolated exactly when the generators of some $\mathfrak p_i$ of its dimension
reduce to zero modulo a Gr\"obner basis of $\sqrt{\mathfrak q}$.

\subsection{Theoretical analysis}
\label{subsec:ideal-branching-analysis}

We need to show that Algorithm~\ref{alg:PrimaryBranching} terminates and
returns a minimal primary decomposition of $\langle P\rangle$ with each
component marked correctly. This will be done by first showing that
Algorithm~\ref{alg:IdealReduce} terminates and keeps $\langle P\rangle$, and
then that the separators give the isolated components.

Lemma~\ref{lem:sat-branches} justifies the powers that
Algorithm~\ref{alg:IdealReduce} adds: each lies in the ideal. Its proof shows
that the cofactors of $h$ have no common zero on $V(Q)$ and then applies the
Nullstellensatz.

\begin{lemma}[SAT branches]
  \label{lem:sat-branches}
  Let $\mathcal H_{\mathrm{sat}}\subseteq\mathcal H(Q)$ contain every SAT
  minimal hitting set, and let $h\in C(Q)$ lie in every
  $H\in\mathcal H_{\mathrm{sat}}$. Then $h^{a_h}\in\langle Q\rangle$.
\end{lemma}

\begin{proof}
  For $q\in Q$ with $h\in C_q$, let $r_q=q/h^{e_{q,h}}$ be the cofactor of $h$.
  We first show that the cofactors have no common zero on $V(Q)$. Suppose that
  $b\in V(Q)$ is a common zero of the cofactors. Then every $q\in Q$ has a
  factor $f\neq h$ with $f(b)=0$: a factor of $r_q$ if $h\in C_q$, and any
  factor that vanishes at $b$ otherwise. So the factors $f\neq h$ in $C(Q)$
  that vanish at $b$ form a hitting set, and a minimal hitting set $H$ inside
  it has $b\in V(H)$ and $h\notin H$. Then $H$ is SAT, hence
  $H\in\mathcal H_{\mathrm{sat}}$ and $h\in H$, a contradiction. By the
  Nullstellensatz, $1=\sum_{q\in Q}g_q\,q+\sum_{q:\,h\in C_q}g'_q\,r_q$ for
  some $g_q,g'_q\in R$. Since $e_{q,h}\leq a_h$, every
  $h^{a_h}r_q=h^{a_h-e_{q,h}}q$ lies in $\langle Q\rangle$, and multiplying by
  $h^{a_h}$ gives $h^{a_h}\in\langle Q\rangle$.
\end{proof}

Algorithm~\ref{alg:IdealReduce} terminates for the same reason as
Algorithm~\ref{alg:Refine} (Theorem~\ref{thm:refine-termination}): $M(Q)$
grows strictly in every round that does not return, since every new power has
a leading monomial outside $M(Q)$.

\begin{theorem}[Termination of ideal reduction]
  \label{thm:ideal-reduce-termination}
  If $\mathcal B$ terminates on every finite polynomial system, then
  Algorithm~\ref{alg:IdealReduce} terminates on every finite polynomial system.
\end{theorem}

\begin{proof}
  Consider a round with $\Delta\neq\emptyset$, and let
  $Q_0=\Delta\cup\{q\in Q:\text{no element of $\Delta$ divides $q$}\}$, so that
  the round ends with $\Call{Interreduce}{Q_0}$. Every $q\in Q$ is kept or is a
  multiple of an element of $\Delta$, so $M(Q)\subseteq M(Q_0)$. The inclusion
  is strict. Indeed, let $\delta=h^{a_h}\in\Delta$, and let $q\in Q$ have
  $e_{q,h}=a_h$, so that the monic $\delta$ divides $q$. Since $Q$ is
  interreduced at the start of every round and $\delta\notin Q$,
  Subsection~\ref{subsec:groebner} gives $\operatorname{LM}(\delta)\notin M(Q)$.
  Since interreduction does not shrink $M$, $M(Q)$ grows strictly in every
  round that does not return. Since $R$ is Noetherian, there are finitely many
  rounds. Each round runs Algorithm~\ref{alg:Branching} with $\mathcal B$ on the
  finitely many $H\in\mathcal H(Q)$, which terminates by
  Theorem~\ref{thm:branching-termination}.
\end{proof}

Using Lemma~\ref{lem:sat-branches}, we show that
Algorithm~\ref{alg:IdealReduce} keeps the ideal.

\begin{theorem}[Ideal reduction]
  \label{thm:ideal-reduce}
  Algorithm~\ref{alg:IdealReduce} terminates and returns a finite
  generating set $Q$ of $\langle P\rangle$.
\end{theorem}

\begin{proof}
  The algorithm terminates by Theorem~\ref{thm:ideal-reduce-termination}.
  Every round keeps $\langle Q\rangle$. Indeed, $\mathcal B$ certifies
  $V(H)=\emptyset$ only when it holds (Lemma~\ref{lem:recursive-refinement}), so
  $\mathcal H_{\mathrm{sat}}$ contains every SAT minimal hitting set. Hence
  $\Delta\subseteq\langle Q\rangle$ by Lemma~\ref{lem:sat-branches}, the dropped
  generators are multiples of elements of $\Delta$, and interreduction keeps the
  ideal.
\end{proof}

% Simplicity: each new power divides the generators it replaces
Like the branches of Algorithm~\ref{alg:Refine}, the output of
Algorithm~\ref{alg:IdealReduce} should be simpler than $P$. Each new power
divides the generators it replaces, which makes every round simpler in the
sense of Theorem~\ref{thm:refine-simpler}.

\begin{theorem}[Reduced systems are simpler]
  \label{thm:ideal-reduce-simpler}
  Let a round of Algorithm~\ref{alg:IdealReduce} with $\Delta\neq\emptyset$
  replace $Q$ by $Q'$. The leading monomials of $Q'$ are smaller than those of
  $Q$ in the multiset extension of $\prec$, yet generate a strictly larger
  monomial ideal. The same holds for the output and $P$, unless the output is
  $\Call{Interreduce}{P}$.
\end{theorem}

\begin{proof}
  Let $Q_0$ be as in the proof of Theorem~\ref{thm:ideal-reduce-termination},
  so that $Q'=\Call{Interreduce}{Q_0}$, and let $D$ be the set of $q\in Q$ that
  some element of $\Delta$ divides. Then $Q_0=\Delta\cup(Q\setminus D)$, and
  $\Delta\cap Q=\emptyset$. Let $\delta=h^{a_h}\in\Delta$, and let $q\in Q$
  have $e_{q,h}=a_h$. Then $\delta\mid q$, so $q\in D$, and $\delta\neq q$,
  since $\delta\notin Q$. Both polynomials are monic, so $q/\delta$ is not
  constant, and $\operatorname{LM}(\delta)\prec\operatorname{LM}(q)$ as in the
  proof of Theorem~\ref{thm:refine-simpler}. Assign each $\delta\in\Delta$ to
  one such $q$. The multiset of the leading monomials of $Q_0$ arises from
  that of $Q$ by replacing $\operatorname{LM}(q)$, for every $q\in D$, with
  the leading monomials of the elements of $\Delta$ assigned to $q$, which are
  finitely many and smaller. Since $\Delta\neq\emptyset$, $D$ is nonempty, so
  the multiset decreases strictly.

  Interreduction does not increase this multiset and does not shrink $M$
  (Subsection~\ref{subsec:groebner}). Hence the multiset of $Q'$ is strictly
  smaller than that of $Q$, and $M(Q)\subsetneq M(Q')$ by the proof of
  Theorem~\ref{thm:ideal-reduce-termination}. The loop starts from
  $\Call{Interreduce}{P}$, whose multiset is at most that of $P$ and whose
  $M$ contains $M(P)$, and it changes $Q$ only in rounds with
  $\Delta\neq\emptyset$. If the output is not $\Call{Interreduce}{P}$, at
  least one such round runs, which gives the second claim.
\end{proof}

The optional lines of Algorithm~\ref{alg:PrimaryBranching} rest on the
following lemma. Saturating by a separator of $\mathfrak p$ keeps only
components whose radicals contain $\mathfrak p$, and among them the
$\mathfrak p$-primary one is the only one of maximal dimension.

\begin{lemma}[Isolated components by separators]
  \label{lem:isolated-separator}
  Let $\mathfrak p$ be a minimal prime of an ideal $I$ and $w$ a separator of
  $\mathfrak p$. Then $\mathfrak p$ is the only minimal prime of $I:w^\infty$,
  and the $\mathfrak p$-primary component of $I$ is the intersection of the
  primary components of $I:w^\infty$ of maximal dimension.
\end{lemma}

\begin{proof}
  Let $I=\mathfrak q_1\cap\cdots\cap\mathfrak q_t$ be a minimal primary
  decomposition with $\mathfrak p_j=\sqrt{\mathfrak q_j}$, and let
  $T=\{j:w\notin\mathfrak p_j\}$. By Subsection~\ref{subsec:decompositions},
  $I:w^\infty=\bigcap_{j\in T}\mathfrak q_j$, and this decomposition is again
  minimal, since a $\mathfrak q_j$ that contained the intersection of the other
  kept components would contain the smaller intersection of all other
  components. For $j\in T$, the prime $\mathfrak p_j$ contains $I$, hence
  $\sqrt I$, and hence a minimal prime of
  $I$~\cite[Prop.~1.11]{atiyah1969introduction}. This minimal prime does not
  contain $w$, so it is $\mathfrak p$. The $\mathfrak p$-primary component
  $\mathfrak q$ of $I$ is kept, since $w\notin\mathfrak p$. So $\mathfrak p$ is
  the only minimal prime of $I:w^\infty$, and every other kept $\mathfrak p_j$
  strictly contains $\mathfrak p$ and therefore has smaller dimension
  (Subsection~\ref{subsec:minimal-prime-decomposition}). Every minimal primary
  decomposition of $I:w^\infty$ has these radicals and the isolated component
  $\mathfrak q$, so $\mathfrak q$ is its only component of maximal dimension.
\end{proof}

Termination of Algorithm~\ref{alg:PrimaryBranching} follows from that of its
steps.

\begin{theorem}[Termination of primary factor branching]
  \label{thm:primary-branching-termination}
  If $\mathcal B$, $\mathcal B_{\min}$, and $\mathcal B_{\mathrm{prim}}$
  terminate on every finite polynomial system, then
  Algorithm~\ref{alg:PrimaryBranching} terminates on every finite polynomial
  system.
\end{theorem}

\begin{proof}
  By Theorem~\ref{thm:ideal-reduce-termination}, Algorithm~\ref{alg:IdealReduce}
  terminates. The remaining lines make finitely many solver calls, each of
  which terminates.
\end{proof}

Finally, the previous results give the correctness of
Algorithm~\ref{alg:PrimaryBranching}. It remains to show that the components
the optional lines return are not redundant.

\begin{theorem}[Primary factor branching]
  \label{thm:primary-branching}
  Algorithm~\ref{alg:PrimaryBranching}, with or without its optional lines,
  terminates and returns a minimal primary decomposition $\mathcal D$ of
  $\langle P\rangle$:
  \[
    \langle P\rangle=\bigcap_{\mathfrak q\in\mathcal D}\mathfrak q.
  \]
  It marks correctly each component isolated or embedded.
\end{theorem}

\begin{proof}
  The algorithm terminates by
  Theorem~\ref{thm:primary-branching-termination}. By
  Theorem~\ref{thm:ideal-reduce}, $\langle Q\rangle=\langle P\rangle=I$, so
  the $\mathfrak p_i$ that $\mathcal B_{\min}$ returns on $Q$ are the minimal
  primes of $I$.

  Suppose that the optional lines return. By
  Lemma~\ref{lem:isolated-separator}, each $\mathfrak q_i$ is the isolated
  $\mathfrak p_i$-primary component of $I$, and
  $\mathfrak q_1\cap\cdots\cap\mathfrak q_r=I$. The radicals $\mathfrak p_i$ are
  distinct, and no $\mathfrak q_i$ is redundant. Otherwise $\mathfrak p_i$
  contains $\bigcap_{j\neq i}\mathfrak q_j$, hence some $\mathfrak q_j$ with
  $j\neq i$~\cite[Prop.~1.11]{atiyah1969introduction}, and hence
  $\mathfrak p_j$, which contradicts the minimality of $\mathfrak p_i$. So the
  $\mathfrak q_i$ form a minimal primary decomposition of $I$, and all of them
  are isolated.

  Otherwise, $\mathcal B_{\mathrm{prim}}$ decomposes $\langle Q\rangle=I$. A
  component is isolated exactly when its radical is a minimal prime of $I$, that
  is, one of the $\mathfrak p_i$, which is the mark.

  In both cases, $g\in\langle P\rangle$ if and only if $g$ lies in every
  component, and a Gr\"obner basis decides membership by reduction
  (Subsection~\ref{subsec:groebner}).
\end{proof}
