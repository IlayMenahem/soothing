\section{Factor branching framework}
\label{sec:method}

% An overview of the section: motivation, what we found, how and why it works
In this section, we present a method that breaks a polynomial system $P$ into
simpler branches, from which a solver describes $V(P)$ and $\sqrt{\langle P
\rangle}$ more easily than from $P$ itself.

\subsection{Framework description}
\label{subsec:branching-framework}
% How breaking down is done
Algorithm~\ref{alg:Refine} breaks the system $P$ into smaller
branches by factoring each generator and enumerating the minimal
hitting sets of the factor sets. Each hitting set generates a new
branch, which we interreduce and factor again, until it reaches a
fixed point. We call a branch \emph{terminal} if one round leaves it unchanged.

\begin{algorithm}[htbp]
  \caption{Refinement}
  \label{alg:Refine}
  \begin{algorithmic}[1]
    \Require A finite polynomial system $P$
    \Ensure The terminal branches of $P$
    \State $W\gets\mathcal H(P)$
    \Repeat
    \State $W_{\mathrm{prev}}\gets W$, \quad $W\gets\emptyset$
    \ForAll{$Q\in W_{\mathrm{prev}}$}
    \State $Q\gets\Call{Interreduce}{Q}$
    \State $W\gets W\cup\mathcal H(Q)$
    \EndFor
    \Until{$W=W_{\mathrm{prev}}$}
    \State $W\gets\Call{Prune}{W}$ \Comment{Optional}
    \State \Return $W$
  \end{algorithmic}
\end{algorithm}

% All about pruning
Hitting sets are minimal only among the children of one branch, so the loop
of Algorithm~\ref{alg:Refine} can end with distinct branches $H$ and $H'$
such that $V(H)\subseteq V(H')$. Solving $H$ then adds no solutions.
$\Call{Prune}{W}$ removes each branch whose solution set it certifies to lie
in that of another kept branch.

% First interreduce?
Interreducing $P$ first can expose factors that its generators hide,
but interreducing can also eliminate factorizations without exposing new ones.
So whether to interreduce first depends on $P$.

% How branching is done
Let $\mathcal L(P)$ denote the terminal branches returned by
Algorithm~\ref{alg:Refine} on $P$.
For each $H\in\mathcal L(P)$, the branch solver must either describe
$V(H)$ exactly or certify that $V(H)=\emptyset$.
We assume that every solver in this paper terminates and returns a
correct output.

\begin{algorithm}[htbp]
  \caption{Factor branching}
  \label{alg:Branching}
  \begin{algorithmic}[1]
    \Require A set of polynomials $P = \{p_1, \ldots, p_m\}$ and a
    branch solver $\mathcal{B}$
    \Ensure A decomposition of $V(P)$

    \State $S \gets \emptyset$ \Comment{Solution-set representations}
    \ForAll{$H\in\Call{Refine}{P}$}
    \State $S \gets S \cup \Call{$\mathcal{B}$}{H}$
    \EndFor

    \State \Return $S$
  \end{algorithmic}
\end{algorithm}

% Optimizations: found primes absorb later branches
A solved branch can prune the later branches whose solution sets it contains.
Branches with fewer generators should then come first: they are likely to be
solved faster and, being likely of higher dimension, to contain later
branches.

% Theoretical analysis of the method
\subsection{Theoretical analysis}
\label{subsec:branching-theoretical-analysis}

We show that Algorithm~\ref{alg:Branching} terminates and returns a
decomposition of $V(P)$ and $\sqrt{\langle P\rangle}$ by first showing that
Algorithm~\ref{alg:Refine} terminates and returns branches whose varieties
have union $V(P)$ and whose radicals intersect to $\sqrt{\langle P\rangle}$.
We then show that these branches are simpler than $P$.

First we need to show that the minimal hitting sets of the factor sets of a
polynomial system $P$ cover its solution set.

\begin{lemma}[One-round branch decomposition]
  \label{thm:branch-decomposition}
  For every finite $Q\subseteq R\setminus\{0\}$,
  \[
    V(Q) = \bigcup_{H \in \mathcal{H}(Q)} V(H).
  \]
\end{lemma}

\begin{proof}
  Let $a\in V(Q)$. Every $q\in Q$ vanishes at $a$, so some factor in $C_q$
  does. The factors in $C(Q)$ that vanish at $a$ therefore form a hitting set,
  which contains a minimal one $H$, and $a\in V(H)$. Conversely, let $a\in V(H)$
  for some $H\in\mathcal H(Q)$. Every $q\in Q$ has a factor in $H$, which
  vanishes at $a$, so $a\in V(Q)$.
\end{proof}

By Lemma~\ref{thm:branch-decomposition}, every round of
Algorithm~\ref{alg:Refine} preserves the solution set, so the terminal
branches describe $V(P)$ and $\sqrt{\langle P\rangle}$.

\begin{lemma}[Refinement preserves solutions]
  \label{lem:recursive-refinement}
  Every round of Algorithm~\ref{alg:Refine} preserves
  $\bigcup_{Q\in W}V(Q)$. Consequently,
  \[
    V(P)=\bigcup_{H\in\mathcal L(P)}V(H)
    \qquad\text{and}\qquad
    \sqrt{\langle P\rangle}=\bigcap_{H\in\mathcal L(P)}\sqrt{\langle H\rangle}.
  \]
\end{lemma}

\begin{proof}
  Interreduction keeps $V(Q)$, so Lemma~\ref{thm:branch-decomposition} gives
  $V(Q)=\bigcup_{H\in\mathcal H(\widetilde Q)}V(H)$ for every branch $Q$, with
  $\widetilde Q=\Call{Interreduce}{Q}$, and a round keeps the union.
  The optional $\Call{Prune}{}$ removes only branches
  whose solution sets lie in that of a kept branch. The loop starts from
  $W=\mathcal H(P)$, so Lemma~\ref{thm:branch-decomposition} and induction on
  the rounds give the first identity. By the Nullstellensatz,
  $\sqrt{\langle P\rangle}=I(V(P))$ and $\sqrt{\langle H\rangle}=I(V(H))$, and
  the vanishing ideal of a finite union is the intersection of the vanishing
  ideals, which gives the second.
\end{proof}

Next we show that Algorithm~\ref{alg:Refine} terminates. First note that each
generator of an interreduced branch is a multiple of an element of each child,
so the leading monomials of a child generate a larger monomial ideal, strictly
unless the child is terminal, and interreducing the child keeps this ideal from
shrinking. Since $R$ is Noetherian, these ideals cannot grow forever, so every
branch eventually becomes terminal, and a round changes no terminal branch.

\begin{theorem}[Termination of refinement]
  \label{thm:refine-termination}
  Algorithm~\ref{alg:Refine} terminates on every finite polynomial system.
\end{theorem}

\begin{proof}
  One round replaces each branch $Q$ by its children
  $\mathcal H(\widetilde Q)$, where $\widetilde Q=\Call{Interreduce}{Q}$,
  independently of the other branches. It leaves a set of terminal branches
  unchanged, and once it leaves $W$ unchanged, so does every later round.

  We first show that every child $H$ of $Q$ is terminal or satisfies
  $M(\widetilde Q)\subsetneq M(\widetilde H)$. Every $q\in\widetilde Q$ is a
  multiple of some $h\in H$, so $\operatorname{LM}(h)\mid\operatorname{LM}(q)$
  and $M(\widetilde Q)\subseteq M(H)$. If equality holds, every $h\in H$ is
  monic, divides some $q\in\widetilde Q$, and has
  $\operatorname{LM}(h)\in M(\widetilde Q)$, so $h=q$
  (Subsection~\ref{subsec:groebner}). Hence $H\subseteq\widetilde Q$, and every
  $q\in\widetilde Q$ equals its factor in $H$, so $H=\widetilde Q$. Then $H$ is
  interreduced and consists of monic irreducible polynomials, so
  $\widetilde H=H$ and $\mathcal H(H)=\{H\}$, that is, $H$ is terminal.
  Otherwise $M(\widetilde Q)\subsetneq M(H)\subseteq M(\widetilde H)$, since
  interreduction does not shrink $M$.

  Next, we show that the rounds started from $\{Q\}$ reach a set of terminal
  branches. Since $R$ is Noetherian, it has no infinite strictly ascending
  chain of ideals, so by Noetherian induction we may assume this for every
  branch $H$ with $M(\widetilde Q)\subsetneq M(\widetilde H)$. A terminal $Q$
  needs no round. Otherwise, each of the finitely many children $H$ of $Q$ is
  terminal or has $M(\widetilde Q)\subsetneq M(\widetilde H)$, so its rounds
  reach terminal branches after some $k_H$ rounds, and those of $\{Q\}$ after
  $1+\max_Hk_H$ rounds.

  Hence, after some $k$ rounds, $W$ consists of terminal branches, and the next
  round leaves it unchanged, so the loop stops. Every later round leaves the
  returned $W$ unchanged, so it equals the work set after round $k$ or later,
  and its branches are terminal.
\end{proof}

Algorithm~\ref{alg:Branching} only calls Algorithm~\ref{alg:Refine} and then
$\mathcal B$ on each branch, so its termination and correctness follow.

\begin{theorem}[Termination of factor branching]
  \label{thm:branching-termination}
  If $\mathcal B$ terminates on every finite polynomial system, then
  Algorithm~\ref{alg:Branching} terminates on every finite polynomial system.
\end{theorem}

\begin{proof}
  By Theorem~\ref{thm:refine-termination}, Algorithm~\ref{alg:Refine}
  terminates and returns a finite set $\mathcal L(P)$, so
  Algorithm~\ref{alg:Branching} makes finitely many calls of $\mathcal B$, each
  of which terminates.
\end{proof}

\begin{theorem}[Factor branching]
  \label{thm:branching-correctness}
  Algorithm~\ref{alg:Branching} terminates and returns a decomposition of
  $V(P)$:
  \[
    V(P)=\bigcup_{H\in S}V(H)
    \qquad\text{and}\qquad
    \sqrt{\langle P\rangle}=\bigcap_{H\in S}\sqrt{\langle H\rangle}.
  \]
\end{theorem}

\begin{proof}
  The algorithm terminates by Theorem~\ref{thm:branching-termination}, and it
  calls $\mathcal B$ on every $H\in\mathcal L(P)$. The branches that
  $\mathcal B$ certifies UNSAT have empty
  solution sets, and $\mathcal B$ represents every other branch exactly, so the
  first identity follows from Lemma~\ref{lem:recursive-refinement}, and the
  second follows from the first by the Nullstellensatz, as in the proof of
  Lemma~\ref{lem:recursive-refinement}.
\end{proof}

% Simplicity: a child keeps at most one irreducible factor of each
% parent generator
Refinement is useful only if its branches are simpler than the system it
starts from. Each element of a child divides a different generator of its
parent, which makes the child a smaller and simpler system in the following
sense.

\begin{theorem}[Refined branches are simpler]
  \label{thm:refine-simpler}
  Let $H\in\mathcal H(\widetilde Q)$ with $H\neq\widetilde Q$. The leading
  monomials of $H$ are smaller than those of $Q$ in the multiset extension of
  $\prec$, yet generate a strictly larger monomial ideal.
\end{theorem}

\begin{proof}
  We first compare $H$ with $\widetilde Q$, then $\widetilde Q$ with $Q$.
  Let $h\in H$. Since $H$ is a minimal hitting set, $H\setminus\{h\}$ is not
  one, so some $q\in\widetilde Q$ has $C_q\cap H=\{h\}$. Let $\iota(h)$ be such
  a $q$. Then $h\mid\iota(h)$, and $\iota$ is injective, since
  $C_{\iota(h)}\cap H=\{h\}$ determines $h$. If $h\neq\iota(h)$, then
  $\iota(h)/h$ is not constant, since both polynomials are monic, so
  $\operatorname{LM}(h)\prec\operatorname{LM}(h)\operatorname{LM}(\iota(h)/h)
  =\operatorname{LM}(\iota(h))$, since $1\prec v$ for every monomial
  $v\neq1$~\cite[Ch.~2, \S4]{cox2015ideals}. The multiset of
  the leading monomials of $H$ arises from that of $\widetilde Q$ by replacing
  $\operatorname{LM}(\iota(h))$ with $\operatorname{LM}(h)$ for every $h\in H$
  and deleting $\operatorname{LM}(q)$ for every
  $q\in\widetilde Q\setminus\iota(H)$. Since $H\neq\widetilde Q$, some $h$ has
  $h\neq\iota(h)$, or some $q$ lies outside $\iota(H)$, so the multiset
  decreases strictly.

  Interreduction does not increase this multiset and does not shrink $M$
  (Subsection~\ref{subsec:groebner}). Hence the multiset of $H$ is strictly
  smaller than that of $Q$, and
  $M(Q)\subseteq M(\widetilde Q)\subsetneq M(H)$, where the strict inclusion is
  shown in the proof of Theorem~\ref{thm:refine-termination}.
\end{proof}

\subsection{Minimal primes}
\label{subsec:minimal-prime-decomposition}

Let $\mathcal B_{\min}$ be a branch solver that returns the minimal
primes of its input. If we run Algorithm~\ref{alg:Branching} with
$\mathcal B_{\min}$ as a branch solver and apply the following
postprocessing, we get the minimal primes of $\langle P\rangle$.

% Postprocessing: keep the returned primes that contain no other
The output $S$ is then the set of primes that $\mathcal B_{\min}$ returns on
the branches in $\mathcal L(P)$, and the postprocessing keeps its minimal
elements under inclusion in two steps.
\begin{enumerate}
  \item Represent each prime by its reduced Gr\"obner basis.
  \item Remove every prime that strictly contains another, testing containment
    by reduction (Subsection~\ref{subsec:groebner}).
\end{enumerate}
A prime that strictly contains $\mathfrak p'$ has smaller dimension than
$\mathfrak p'$, so the second step tests $\mathfrak p'\subseteq\mathfrak p$
only when $\mathfrak p'$ has the larger dimension.

% Correctness
\begin{theorem}[Prime factor branching]
  \label{thm:prime-branching}
  With the branch solver $\mathcal B_{\min}$, the output $S$ of
  Algorithm~\ref{alg:Branching} after the postprocessing is the minimal primes
  of $\langle P\rangle$.
\end{theorem}

\begin{proof}
  By Theorem~\ref{thm:refine-termination}, $\mathcal L(P)$ is finite, so
  $S$ is finite. On each $H\in\mathcal L(P)$, $\mathcal B_{\min}$
  returns the minimal primes of $\langle H\rangle$, which intersect to
  $\sqrt{\langle H\rangle}$, so Lemma~\ref{lem:recursive-refinement} gives
  \[
    \bigcap_{\mathfrak p\in S}\mathfrak p
    =\bigcap_{H\in\mathcal L(P)}\sqrt{\langle H\rangle}
    =\sqrt{\langle P\rangle}.
  \]
  Hence every element of $S$ is a prime containing $\langle P\rangle$, and
  every prime containing $\langle P\rangle$ contains
  $\sqrt{\langle P\rangle}$, hence some element of $S$ and some minimal prime
  of $\langle P\rangle$~\cite[Prop.~1.11]{atiyah1969introduction}. So a
  minimal prime $\mathfrak q$ of $\langle P\rangle$ contains an element of $S$
  and, by its minimality, equals every element of $S$ that it contains; hence
  $\mathfrak q$ is a minimal element of $S$. Conversely, a minimal element of
  $S$ contains a minimal prime of $\langle P\rangle$, which lies in $S$, so the
  two are equal.
\end{proof}
