% the algorithm pseudocode

\begin{algorithm}
\caption{Branching}
\label{alg:Branching}
\begin{algorithmic}[1]
    \Require A set of polynomials $P = \{p_1, \ldots, p_m\}$.

    \State Interreduce the polynomials \Comment{Optional}
    \State Initialize $C_1, \ldots, C_m$ as the sets of factors of $p_1, \ldots, p_m$

    \State Initialize $C \gets \emptyset$ an unsatisfiable cores set
    \State Initialize $S \gets \emptyset$ a solutions set
    \State Initialize $T \gets \emptyset$ a timeout set

    \For{Each minimal hitting set $HS$ of $C_1, \ldots, C_m$}
        \If{$HS$ contains an unsatisfiable core from $C$}
            \State \textbf{continue}
        \EndIf

        \State Check if $HS$ is satisfiable, and check if it is unsatisfiable
        \If{$HS$ is satisfiable}
            \State Add the solution to $S$
        \ElsIf{$HS$ is unsatisfiable}
            \State $U \gets \Call{GBCoreIterativeRefinement}{HS}$ as in Section 5 of \cite{sun2016}
            \State $C \gets C \cup \{U\}$
        \Else
            \State Add $HS$ to $T$
        \EndIf
    \EndFor

    \Return $S$, $C$, $T$

\end{algorithmic}
\end{algorithm}
