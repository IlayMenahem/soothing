\section{Recursive splitting}
\label{sec:variable-zeroing}

\subsection{Factor branching with variable zeroing}
\label{subsec:zeroing-branching}

% Motivation for splitting
Suppose the solver does not terminate within a reasonable amount of time on a
branch produced by Algorithm~\ref{alg:Branching}.

% What is splitting
Then we can further simplify the system by recursive splitting. In each child
branch, we assume a side condition and run Algorithm~\ref{alg:Branching} on the
simplified system. This process continues until every branch is solved or a
global time budget is exhausted.

% Good properties for conditions on the set of conditions
For completeness, the side conditions must cover all possibilities. They
should also simplify the system for the solver input and work well with
Algorithm~\ref{alg:Refine}.

% What is the condition we use, and why?
In this work, we use \emph{variable zeroing}: recursive splitting with the
side conditions $x=0$ and $x\neq0$ for a selected variable $x$. Setting $x=0$
deletes every monomial containing $x$, which can make generators factor and
so create new hitting sets. Requiring $x\neq0$ removes the components of the
branch that lie in $\{x=0\}$.

% Pseudocode for the method with variable zeroing
\begin{algorithm}[htbp]
\small
\caption{Factor branching with variable zeroing}
\label{alg:ZeroingBranching}
\begin{algorithmic}[1]
    \Require A polynomial system $P$, a branch solver $\mathcal B$, a per-call timeout $\tau$, and a global budget $\beta$
    \State $S\gets\emptyset$; $U\gets\emptyset$ \Comment{Solved and unresolved branches}
    \State $W\gets\{(P,\emptyset,\emptyset)\}$ \Comment{Pending branches $(Q,Z,N)$}
    \While{$W\neq\emptyset$}
        \State Remove an element $(Q,Z,N)$ from $W$
        \ForAll{$H\in\Call{Refine}{Q}$}
            \State Run $\mathcal B$ on $H$ for at most $\tau$, unless $\beta$ is exhausted
            \State $V\gets \operatorname{vars}(H)\setminus(Z\cup N)$ \Comment{Useful variables for $H$}
            \If{$\mathcal B$ returns a representation of $V(H)$}
                \State Add the representation, with $(Z,N)$, to $S$
            \ElsIf{$\beta$ is exhausted or $V=\emptyset$}
                \State $U\gets U\cup\{(H,Z,N)\}$
            \Else
                \State $x\gets\Call{ChooseVariable}{H,V}$
                \State $W\gets W\cup\{(H|_{x=0},Z\cup\{x\},N)\}$ \Comment{$x=0$}
                \State $W\gets W\cup\{(\Call{UpdateSaturation}{H,N\cup\{x\}},Z,N\cup\{x\})\}$ \Comment{$x\neq0$}
            \EndIf
        \EndFor
    \EndWhile
    \State \Return $(S,U)$
\end{algorithmic}
\end{algorithm}


% How the constraints are encoded into the system.
Each pending branch $(Q,Z,N)$ records the variables $Z$ set to zero and the
variables $N$ declared nonzero on its path.
Subsection~\ref{subsec:nonzero-encoding} describes two ways to encode the
nonzero conditions, and some solvers have their own internal encoding.

\begin{lemma}
\label{lem:auxiliary-generators}
Refinement keeps exactly one generator containing $s_x$, of the form
$c(1-s_xh_x)$ with $h_x\equiv x$ modulo the auxiliary-free generators $H'$.
Hence $\pi(V(H))=\{a\in V(H'):a_x\neq0,\ x\in N\}$, and
$\mathrm{UpdateSaturation}$ loses no constraint.
\end{lemma}

\begin{proof}
The leading monomial of $1-s_xh_x$ contains $s_x$, so it reduces no other
generator, is reduced only by $H'$ (changing $h_x$ by an element of
$\langle H'\rangle$), and, being linear in $s_x$, does not factor.\qed
\end{proof}

Solutions of the nonzero branch are projected onto the original variables, so
the auxiliary variables do not appear in the output. The zero and nonzero
decisions remain part of the solution-set representation.

For comparison, we repeat the experiment twice: once with the fixed side
conditions $x=1$ and $x\neq1$, and once with each split independently drawing
an integer $a$ uniformly from $\{-10,\ldots,10\}$ and using the side conditions
$x=a$ and $x\neq a$. Both runs have a ten-minute wall-clock limit. Neither
alternative finishes within ten minutes, and in both cases the search has
branches that remain unresolved even after exhausting every variable.

These comparisons demonstrate that zero/nonzero splits are substantially more
effective than arbitrary affine splittings on these structured systems.

\subsection{Choice of variable for splitting}
\label{subsec:zeroing-variable-choice}

The choice of variable for splitting affects the efficiency of the algorithm.
If a variable barely occurs in the current branch, its child branches may
repeat the same work. A variable that occurs in many generators is a more
promising choice. Setting it to zero removes all terms containing it at once.
In the nonzero branch, $\mathrm{UpdateSaturation}$ adds the equation
$\sigma_x=0$.

For each $x\in X = \operatorname{vars}(Q)\setminus(Z\cup N)$, the variables not yet split on, define
\begin{align*}
    g_Q(x) &= \#\{q\in Q : x\text{ occurs in }q\},\\
    t_Q(x) &= \#\{(q,m) : q\in Q,\ m\text{ is a term of }q,\ \deg_x(m) > 0\},\\
    e_Q(x) &= \sum_{q\in Q}\ \sum_{\substack{m\text{ a term of }q,\\ \deg_x(m) > 0}} \deg_x(m).
\end{align*}
Its score is the triple
\[
    h_Q(x)=\bigl(g_Q(x),t_Q(x),e_Q(x)\bigr),
\]
and scores are compared lexicographically.
$\mathrm{ChooseVariable}(Q,X)$ returns a variable $x\in X$ with the largest score $h_Q(x)$.

\subsection{Branch contraction}
\label{subsec:zeroing-branch-contraction}

Suppose that one child of a split on $x$ has no terminal branches (i.e., UNSAT) and the other has exactly one, $H$.
Lemma~\ref{lem:recursive-refinement} and Equation~\eqref{eq:zeroing-split} give
\[
    V_Y(Q;Z,N)=
    \begin{cases}
        V_Y(H;Z,N\cup\{x\}) & \text{if the $x=0$ child is UNSAT,}\\
        V_Y(H;Z\cup\{x\},N) & \text{if the $x\neq0$ child is UNSAT.}
    \end{cases}
\]
Thus we can choose the next splitting variable without running the branch solver on $H$, or losing solutions.
This optimization may save a solver call that is likely to fail.

\subsection{Theoretical analysis}
\label{subsec:zeroing-theoretical-analysis}

The correctness argument uses only that the side conditions cover all
possibilities. Let $Y=\operatorname{vars}(P)$ be the original variables, let
$A$ be the conditions on the current branch, and let $V_Y(Q;A)$ be the
projection onto $Y$ of the solutions of $Q$ satisfying $A$. For variable
zeroing, a pending branch $(Q,Z,N)$ has
$Q\subseteq K[Y\setminus Z][s_y:y\in N]$, and we write
\[
    V_Y(Q;Z,N)=\bigl\{a\in\overline K^{Y}:
    a_z=0\ (z\in Z),\ a_y\neq0\ (y\in N),\
    (a|_{Y\setminus Z},s)\in V(Q)\text{ for some }s\bigr\}.
\]
At the root, $V_Y(P;\emptyset,\emptyset)=V(P)$.

\begin{theorem}[Correctness of an exhaustive split]
\label{lem:zeroing-split}
Let $D_1,\ldots,D_k$ be side conditions such that every point satisfying $A$
satisfies at least one $D_i$. If each $Q_i$ satisfies
\[
    V_Y(Q_i;A\cup\{D_i\})=\{a\in V_Y(Q;A):a\text{ satisfies }D_i\},
\]
then
\[
    V_Y(Q;A)=\bigcup_{i=1}^k V_Y(Q_i;A\cup\{D_i\}).
\]
\end{theorem}

\begin{proof}
Each set on the right is contained in $V_Y(Q;A)$ by hypothesis. Conversely,
a point $a\in V_Y(Q;A)$ satisfies $A$, hence some $D_i$, and so lies in the
$i$-th set on the right.\qed
\end{proof}

For variable zeroing, the conditions $x=0$ and $x\neq0$ cover all values of
$x$.

\begin{lemma}
\label{lem:zero-nonzero-split}
Let $H$ be a terminal branch of a pending branch $(Q,Z,N)$, let
$Q_0=H|_{x=0}$, and let $Q_1=\mathrm{UpdateSaturation}(H,N\cup\{x\})$. Then
\begin{equation}
\label{eq:zeroing-split}
    V_Y(H;Z,N)=V_Y(Q_0;Z\cup\{x\},N)
    \cup V_Y(Q_1;Z,N\cup\{x\}).
\end{equation}
\end{lemma}

\begin{proof}
By Theorem~\ref{lem:zeroing-split}, it suffices to show that $Q_0$ and $Q_1$
satisfy its hypothesis. For $Q_0$, a generator $q$ satisfies
$q|_{x=0}(b,s)=q(b,s)$ whenever $b_x=0$, so a point lies in
$V_Y(Q_0;Z\cup\{x\},N)$ exactly when it lies in $V_Y(H;Z,N)$ and has $a_x=0$.
For $Q_1$, let $H'$ be the auxiliary-free generators of $H$.
Lemma~\ref{lem:auxiliary-generators} and
Subsection~\ref{subsec:nonzero-encoding} give
$\pi(V(H))=\{b\in V(H'):b_y\neq0,\ y\in N\}$ and
$\pi(V(Q_1))=\{b\in V(H'):b_y\neq0,\ y\in N\cup\{x\}\}$, so a point lies in
$V_Y(Q_1;Z,N\cup\{x\})$ exactly when it lies in $V_Y(H;Z,N)$ and has
$a_x\neq0$.\qed
\end{proof}

Lemma~\ref{lem:recursive-refinement}, applied in the ring of $Q$, gives
$V(Q)=\bigcup_{H\in\mathcal L(Q)}V(H)$. Since $V_Y(Q;Z,N)$ depends on $Q$ only
through $V(Q)$,
\begin{equation}
\label{eq:zeroing-refinement}
    V_Y(Q;Z,N)=\bigcup_{H\in\mathcal L(Q)}V_Y(H;Z,N).
\end{equation}

\begin{theorem}[Correctness of recursive splitting]
\label{thm:zeroing-correctness}
Suppose every split uses side conditions that cover all possibilities. If
Algorithm~\ref{alg:ZeroingBranching} returns $(S,U)$ on input $P$, then
\[
    V(P)=\operatorname{Sol}(S)
    \cup\bigcup_{(H,Z,N)\in U}V_Y(H;Z,N),
\]
where a representation in $S$ stored with $(Z,N)$ for the branch $H$
represents $V_Y(H;Z,N)$.
\end{theorem}

\begin{proof}
We show that the loop of Algorithm~\ref{alg:ZeroingBranching} preserves the
invariant
\[
    V(P)=\operatorname{Sol}(S)
    \cup\bigcup_{(H,Z,N)\in U}V_Y(H;Z,N)
    \cup\bigcup_{(Q,Z,N)\in W}V_Y(Q;Z,N).
\]
Before the loop, $S=U=\emptyset$ and $W=\{(P,\emptyset,\emptyset)\}$, so the
invariant reads $V(P)=V_Y(P;\emptyset,\emptyset)$.

One iteration removes $(Q,Z,N)$ from $W$. By
Equation~\eqref{eq:zeroing-refinement}, $V_Y(Q;Z,N)$ is the union of
$V_Y(H;Z,N)$ over $H\in\mathcal L(Q)$, and the iteration places each such
set elsewhere. If $\mathcal B$ returns a representation of $V(H)$, the
algorithm adds it with $(Z,N)$ to $S$, so it represents $V_Y(H;Z,N)$. If
$\mathcal B$ certifies $H$ as UNSAT, then $V_Y(H;Z,N)=\emptyset$. If $H$ is
unresolved and not split, $(H,Z,N)$ enters $U$. Otherwise the two children
enter $W$, and their sets have union $V_Y(H;Z,N)$ by
Equation~\eqref{eq:zeroing-split}. Hence the invariant still holds.

When the algorithm returns, $W=\emptyset$, and the invariant is the claimed
equality.\qed
\end{proof}

\begin{corollary}[Completeness of variable zeroing]
\label{cor:zeroing-completeness}
If $U=\emptyset$ in the output of Algorithm~\ref{alg:ZeroingBranching}, then
$\operatorname{Sol}(S)=V(P)$.
\end{corollary}
