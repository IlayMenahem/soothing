\begin{tikzpicture}[
    font=\small,
    every node/.style={align=center},
    blue box/.style={draw=blue!70!cyan, line width=1pt, rounded corners=6pt,
        fill=blue!50!cyan!4, inner sep=7pt},
    loop box/.style={draw=orange!85!black, line width=1pt, rounded corners=6pt,
        fill=yellow!40!orange!12, inner sep=7pt},
    decision/.style={draw=green!55!black, line width=1pt, diamond, aspect=1.8,
        fill=green!60!teal!8, inner sep=1pt},
    green box/.style={draw=green!55!black, line width=1pt, rounded corners=6pt,
        fill=green!60!teal!4, inner sep=7pt},
    red box/.style={draw=red!70, line width=1pt, rounded corners=6pt,
        fill=red!4, inner sep=7pt},
    return box/.style={draw=violet, line width=1pt, rounded corners=6pt,
        fill=violet!6, inner sep=7pt},
    arr/.style={-{Stealth[length=2.6mm]}, line width=1pt},
]

% ---- initialization and refinement ----
\node[blue box, text width=43mm] (init) {%
    \textcolor{blue}{\bfseries Initialize}\\
    $P = \{p_1,\ldots,p_m\}$\\
    $S \leftarrow \emptyset$};

\node[blue box, text width=45mm, right=8mm of init] (factors) {%
    \textcolor{blue}{\bfseries Refine}\\
    Factor, then interreduce\\ and refactor to obtain $\mathcal L(P)$};

% ---- loop over terminal branches ----
\node[loop box, below=9mm of factors, xshift=-22mm] (loop) {%
    \textcolor{orange!75!black}{\bfseries Next set}\\
    $H\in\mathcal L(P)$};

% ---- solve ----
\node[blue box, below=6mm of loop] (solve)
    {\textcolor{blue}{\bfseries Solve $H$}};

% ---- two outcome branches ----
\node[red box, anchor=west] (unsat) at ($(solve.south)+(8mm,-10mm)$) {%
    \textcolor{red}{\bfseries UNSAT}\\
    Certify $V(H)=\emptyset$};

\node[green box, anchor=east] (sat) at ($(solve.south)+(-8mm,-10mm)$) {%
    \textcolor{green!45!black}{\bfseries SAT}\\
    Add solutions to $S$};

% ---- bottom rail ----
\coordinate (raill) at ($(sat.south)+(0,-4mm)$);
\coordinate (railm) at (raill -| solve.south);
\coordinate (railr) at (raill -| unsat.south);

% ---- decision and return ----
\node[decision] (more) at ($(railm)+(0,-10mm)$) {More $H$?};
\node[return box, below=7mm of more] (return)
    {\textcolor{violet}{\bfseries Return $S$}};

% ---- arrows: top pipeline ----
\draw[arr] (init.east) -- (factors.west);
\coordinate (entry) at ($(loop.north)+(0,4mm)$);
\draw[arr, rounded corners=5pt]
    (factors.south) |- (entry) -- (loop.north);

% ---- loop body ----
\draw[arr] (loop.south) -- (solve.north);
\coordinate (outcome split) at ($(solve.south)+(0,-6mm)$);
\draw[line width=1pt] (solve.south) -- (outcome split);
\draw[arr] (outcome split)
    .. controls +(0,-2.2mm) and +(-2.2mm,0) .. ++(4mm,-4mm)
    -- (unsat.west);
\draw[arr] (outcome split)
    .. controls +(0,-2.2mm) and +(2.2mm,0) .. ++(-4mm,-4mm)
    -- (sat.east);

% ---- bottom rail into the decision ----
\draw[line width=1pt] (sat.south) -- (raill) -- (railr) -- (unsat.south);
\draw[arr] (railm) -- (more);
\draw[arr] (more) -- node[right] {no} (return);

% ---- feedback: another branch re-enters the loop ----
\coordinate (fbcol) at ($(sat.west)+(-8mm,0)$);
\coordinate (feedback) at (fbcol |- loop.west);
\draw[line width=1pt, rounded corners=5pt]
    (more.west) -- node[below] {yes} (more.west -| fbcol) -- (feedback);
\draw[arr] (feedback) -- (loop.west);

\end{tikzpicture}
