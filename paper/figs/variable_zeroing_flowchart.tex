\begin{tikzpicture}[
    font=\small,
    every node/.style={align=center},
    blue box/.style={draw=blue!70!cyan, line width=1pt, rounded corners=6pt,
        fill=blue!50!cyan!4, inner sep=7pt},
    loop box/.style={draw=orange!85!black, line width=1pt, rounded corners=6pt,
        fill=yellow!40!orange!12, inner sep=7pt},
    decision/.style={draw=green!55!black, line width=1pt, diamond, aspect=1.8,
        fill=green!60!teal!8, inner sep=1pt},
    green box/.style={draw=green!55!black, line width=1pt, rounded corners=6pt,
        fill=green!60!teal!4, inner sep=7pt},
    red box/.style={draw=red!70, line width=1pt, rounded corners=6pt,
        fill=red!4, inner sep=7pt},
    return box/.style={draw=violet, line width=1pt, rounded corners=6pt,
        fill=violet!6, inner sep=7pt},
    connector/.style={line width=1pt, line cap=round, line join=round},
    arr/.style={connector, -{Stealth[length=2.6mm, width=2mm]}},
]

\node[blue box] (init) {%
    \textcolor{blue}{\bfseries Initialize}\\
    $X\leftarrow\Call{Variables}{P}$\\
    $\Call{SolveByZeroing}{P,X,\emptyset,\emptyset,\tau}$};

\node[blue box, below=7mm of init] (solve) {%
    $(S,T)\leftarrow\Call{FactorBranchingWithTimeout}{Q,\tau}$};
\node[decision, below=7mm of solve] (next) {Unresolved\\$R\in T$?};
\node[return box, left=20mm of next] (return) {%
    \textcolor{violet}{\bfseries Return}\\$(S,U)$};

\node[loop box, below=8mm of next] (choose) {%
    \textcolor{orange!75!black}{\bfseries Process $R$}\\
    If $X=\emptyset$, add $(R,Z,N)$ to $U$\\
    otherwise choose $x\in X$ and split};

\node[blue box] (zero) at ($(choose.south)+(-40mm,-20mm)$) {%
    \textcolor{blue}{\bfseries Zero branch}\\
    $R_0\leftarrow R[x\mapsto0]$\\
    Recurse with $Z\cup\{x\}$};

\node[blue box] (nonzero) at ($(choose.south)+(40mm,-20mm)$) {%
    \textcolor{blue}{\bfseries Nonzero branch}\\
    $R_1\leftarrow\Call{UpdateSaturation}{R,N\cup\{x\}}$\\
    Recurse with $N\cup\{x\}$};

\node[green box] (merge) at ($(choose.south)+(0,-46mm)$) {%
    Add both child results to $S$ and $U$};

\draw[arr] (init) -- (solve);
\draw[arr] (solve) -- (next);
\draw[arr] (next.west) -- node[above] {No} (return.east);
\draw[arr] (next) -- node[right] {Yes} (choose);

\coordinate (split) at ($(choose.south)+(0,-4mm)$);
\draw[connector] (choose.south) -- (split);
\draw[arr] (split) -- (zero.north);
\draw[arr] (split) -- (nonzero.north);
\draw[arr, rounded corners=4pt] (zero.south) |- (merge.west);
\draw[arr, rounded corners=4pt] (nonzero.south) |- (merge.east);

\coordinate (rightcol) at ($(nonzero.east)+(10mm,0)$);
\coordinate (belowmerge) at ($(merge.south)+(0,-4mm)$);
\draw[arr, rounded corners=4pt]
    (merge.south) -- (belowmerge) -| (rightcol) |- (next.east);

\end{tikzpicture}
