\section{Method}

% some background on the method
Let us say we have a very large polynomial system $P$, we factorize the polynomials in $P$ and get a set of factors $C_1, \ldots, C_m$.
We iterate over all the minimal hitting sets of $C_1, \ldots, C_m$, and solve the polynomial system corresponding to each minimal hitting set.
We show that the union of the solution sets our algorithm returns is the full solution set of $P$.
Additionally, we optimize the order of iteration over the minimal hitting sets, and learn unsatisfiable cores to skip hitting sets that contain them and thus unsolvable.

\begin{algorithm}
\caption{Branching}
\label{alg:Branching}
\begin{algorithmic}[1]
    \Require A set of polynomials $P = \{p_1, \ldots, p_m\}$.

    \State Interreduce the polynomials \Comment{Optional}
    \State Initialize $C_1, \ldots, C_m$ as the sets of factors of $p_1, \ldots, p_m$

    \State Initialize $C \gets \emptyset$ an unsatisfiable cores set
    \State Initialize $S \gets \emptyset$ a solutions set

    \For{Each minimal hitting set $HS$ of $C_1, \ldots, C_m$}
        \If{$HS$ contains an unsatisfiable core from $C$}
            \State \textbf{continue}
        \EndIf

        \State Check whether $HS$ is satisfiable
        \If{$HS$ is satisfiable}
            \State Add the solutions returned for $HS$ to $S$
        \Else
            \State $C \gets C \cup \{\Call{GBCoreIterativeRefinement}{HS}\}$ as in Section 5 of \cite{sun2016}
        \EndIf
    \EndFor

    \Return $S$, $C$

\end{algorithmic}
\end{algorithm}

\subsection{Iteration over minimal hitting sets}

There are some additional details to consider so the iteration over the minimal hitting sets is efficient.

\paragraph*{Order of iteration:} The order of iteration over the hitting sets is from the simplest to the most complex to check for satisfiability.
As a heuristic for this cost we use the rule more constraint factors, the more complex to check.
This allows our method to learn early unsatisfiable cores that are easier to check and prune the search space.

\paragraph*{Lazy iteration:} The algorithm iterates over the hitting sets in a lazy manner, skipping hitting sets that contain a learned unsatisfiable core.

\subsection{Theoretical analysis}

\begin{lemma}[Correctness of interreduction]
\label{lem:interreduction}
Suppose that one generator $p_i$ is replaced by
\[
    p_i' = p_i-\sum_{j\neq i} a_jp_j,
    \qquad a_j\in K[x_1,\ldots,x_n].
\]
Then
\[
    \langle p_1,\ldots,p_i,\ldots,p_m\rangle
    =
    \langle p_1,\ldots,p_i',\ldots,p_m\rangle.
\]
Any finite sequence of these reductions therefore preserves the generated
ideal and its zero set.
\end{lemma}

\begin{proof}
The polynomial $p_i'$ belongs to the original ideal, while
$p_i=p_i'+\sum_{j\neq i}a_jp_j$ belongs to the ideal generated by the updated
system. Each generating set therefore generates the other. The same argument
applies at every step of a sequential interreduction.
\end{proof}

A zero generator imposes no constraint and can be discarded. A nonzero
constant makes the ideal the unit ideal, so the input system is unsatisfiable.

\begin{lemma}[Sufficiency of minimal hitting sets]
\label{lem:minimal-hitting-suffices}
Every hitting set $T$ of the factor family contains a minimal hitting set
$H\subseteq T$, with $V(T)\subseteq V(H)$. It is therefore enough to consider
minimal hitting sets.
\end{lemma}

\begin{proof}
Because the factor sets are finite, repeatedly removing redundant factors from
$T$ eventually produces a minimal hitting set $H$. Since $H\subseteq T$, any
point in $V(T)$ also lies in $V(H)$.
\end{proof}

\begin{theorem}
\label{thm:branch-decomposition}
The solution set of $P$ is
\[
    V(P) = \bigcup_{H \in \mathcal{H}} V(H).
\]
\end{theorem}

\begin{proof}
Take $x \in V(P)$. Since $p_i(x)=0$, at least one factor in $C_i$ vanishes at
$x$ for every $i$. The set of all factors that vanish at $x$ is therefore a
hitting set of $C_1,\ldots,C_m$. Because the factor sets are finite, this set
contains a minimal hitting set $H$. Every factor in $H$ vanishes at $x$, so
$x \in V(H)$.

For the reverse inclusion, take $x \in V(H)$ for some $H \in \mathcal{H}$.
The set $H$ intersects every $C_i$, so each $p_i$ has a factor that vanishes at
$x$. Hence $p_i(x)=0$ for every $i$, and $x \in V(P)$.
\end{proof}

The same decomposition can be stated in terms of ideals.

\begin{theorem}[Radical decomposition]
\label{thm:radical-decomposition}
The radical of the input ideal satisfies
\[
    \sqrt{\langle P\rangle}
    =
    \bigcap_{H\in\mathcal H}\sqrt{\langle H\rangle}.
\]
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:branch-decomposition},
$V(P)=\bigcup_{H\in\mathcal H}V(H)$. Taking vanishing ideals gives
\[
    I(V(P))
    = I\left(\bigcup_{H\in\mathcal H}V(H)\right)
    = \bigcap_{H\in\mathcal H} I(V(H)).
\]
Hilbert's Nullstellensatz identifies $I(V(P))$ with
$\sqrt{\langle P\rangle}$ and each $I(V(H))$ with
$\sqrt{\langle H\rangle}$, which gives the stated equality.
\end{proof}

\begin{lemma}
\label{lem:core-pruning}
Let $U$ be an unsatisfiable core. If $U \subseteq H$, then $V(H)=\emptyset$.
\end{lemma}

\begin{proof}
Since every point in $V(H)$ must satisfy every polynomial in $U$, we have
$V(H) \subseteq V(U)$. The core $U$ is unsatisfiable, so $V(U)=\emptyset$ and
therefore $V(H)=\emptyset$.
\end{proof}

Soundness does not require $U$ to be minimal. A smaller core is useful only
because more later branches may contain it.

For the following result, assume that the branch solver terminates on every
unpruned branch $H$. If $H$ is satisfiable, it returns exactly $V(H)$. If $H$
is unsatisfiable, it returns a certified core $U\subseteq H$.

\begin{theorem}
\label{thm:branching-correctness}
Algorithm~\ref{alg:Branching} is sound and complete for $P$.
\end{theorem}

\begin{proof}
Let $P'$ denote the system after the optional interreduction step. We check
that each part of the algorithm preserves the solution set.

\paragraph*{Interreduction.}
Lemma~\ref{lem:interreduction} gives $\langle P'\rangle=\langle P\rangle$, and hence $V(P')=V(P)$.

\paragraph*{Factorization and branch construction.}
Theorem~\ref{thm:branch-decomposition}, applied to the factor sets of $P'$,
gives
\[
    V(P')=\bigcup_{H\in\mathcal H}V(H).
\]
Lemma~\ref{lem:minimal-hitting-suffices} shows that no nonminimal hitting set
can add a solution missing from this union. Thus the initial branch collection
represents exactly $V(P')$.

\paragraph*{Loop invariant.}
Let $R$ be the set of branches that have not yet been processed, and let $S$
be the solution set accumulated so far. At the start of the loop,
\[
    S\subseteq V(P')
    \qquad\text{and}\qquad
    V(P')\subseteq S\cup\bigcup_{H\in R}V(H).
\]
The first inclusion is the soundness invariant. The second is the completeness
invariant: every solution not yet in $S$ remains in a pending branch. Both
hold initially because $S=\emptyset$, $R=\mathcal H$, and the branch
decomposition holds.

\paragraph*{Core pruning.}
Suppose the algorithm skips $H$ because it contains a learned core $U$.
Lemma~\ref{lem:core-pruning} gives $V(H)=\emptyset$. Removing $H$ from $R$
therefore leaves the completeness invariant unchanged, so soundness is unchanged as well.

\paragraph*{Branch solving.}
If $H$ is satisfiable, the solver adds exactly $V(H)$ to $S$. Since
$V(H)\subseteq V(P')$, the soundness invariant is preserved. Moving $V(H)$
from the pending union into $S$ also preserves the completeness invariant. If
$H$ is unsatisfiable, then $V(H)=\emptyset$, so removing it from $R$ preserves
both invariants.

\paragraph*{Return.}
When the loop ends, $R=\emptyset$. The completeness invariant gives
$V(P')\subseteq S$, while the soundness invariant gives $S\subseteq V(P')$.
Hence $S=V(P')=V(P)$.
\end{proof}

Note that the theoretical analysis does not require the field $K$ to be algebraically closed.
This is due to if there is no solution over the algebraic closure, then there is no solution over the field itself.
