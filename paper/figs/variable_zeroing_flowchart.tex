\begin{tikzpicture}[
    font=\small,
    every node/.style={align=center},
    blue box/.style={draw=blue!70!cyan, line width=1pt, rounded corners=6pt,
        fill=blue!50!cyan!4, inner sep=7pt},
    loop box/.style={draw=orange!85!black, line width=1pt, rounded corners=6pt,
        fill=yellow!40!orange!12, inner sep=7pt},
    decision/.style={draw=green!55!black, line width=1pt, diamond, aspect=1.8,
        fill=green!60!teal!8, inner sep=1pt},
    green box/.style={draw=green!55!black, line width=1pt, rounded corners=6pt,
        fill=green!60!teal!4, inner sep=7pt},
    red box/.style={draw=red!70, line width=1pt, rounded corners=6pt,
        fill=red!4, inner sep=7pt},
    return box/.style={draw=violet, line width=1pt, rounded corners=6pt,
        fill=violet!6, inner sep=7pt},
    arr/.style={-{Stealth[length=2.6mm]}, line width=1pt},
]

% ---- initialization and entry into the recursive routine ----
\node[blue box] (init) {%
    \textcolor{blue}{\bfseries Initialize}\\
    $(P,\tau)$, $C\leftarrow\emptyset$, $X\leftarrow\Call{Variables}{P}$};

\node[blue box, right=10mm of init] (entry) {%
    $\Call{SolveByZeroing}{P,X,\emptyset,\emptyset,\tau,C}$};

% ---- one invocation of SolveByZeroing ----
\node[blue box, below=8mm of entry] (solve) {%
    $(S,T)\leftarrow\Call{BranchingWithTimeout}{Q,\tau,C}$};

\node[blue box, below=5mm of solve] (unresolved) {$U\leftarrow\emptyset$};

\node[decision, below=6mm of unresolved] (next) {Unprocessed\\$Q\in T$?};

\node[return box, left=18mm of next] (return) {%
    \textcolor{violet}{\bfseries Return}\\
    $(S,U,C)$};

\node[decision, below=7mm of next] (variables) {Splitting\\variable?};

\node[red box, right=16mm of variables] (keep) {%
    Add $(Q,Z,N)$ to $U$};

\node[loop box, below=7mm of variables] (choose) {%
    \textcolor{orange!75!black}{\bfseries Split}\\
    Choose $x\in X$};

% ---- the two recursive children ----
\node[blue box] (zero) at ($(choose.south)+(-38mm,-17mm)$) {%
    \textcolor{blue}{\bfseries $x=0$}\\
    $Q\leftarrow Q[x\mapsto0]$\\
    $Z\leftarrow Z\cup\{x\}$};

\node[blue box] (nonzero) at ($(choose.south)+(38mm,-17mm)$) {%
    \textcolor{blue}{\bfseries $x\ne0$}\\
    Update saturation\\
    $N\leftarrow N\cup\{x\}$};

\node[green box] (merge) at ($(choose.south)+(0,-43mm)$) {%
    Merge child results};

% ---- arrows ----
\draw[arr] (init) -- (entry);
\draw[arr] (entry.south) -- (solve.north);
\draw[arr] (solve) -- (unresolved);
\draw[arr] (unresolved) -- (next);

\draw[arr] (next.west) -- node[above] {No} (return.east);
\draw[arr] (next.south) -- node[right] {Yes} (variables.north);
\draw[arr] (variables.east) -- node[above] {No} (keep.west);
\draw[arr] (variables.south) -- node[right] {Yes} (choose.north);

\coordinate (split) at ($(choose.south)+(0,-4mm)$);
\draw[line width=1pt] (choose.south) -- (split);
\draw[arr] (split) -- (zero.north);
\draw[arr] (split) -- (nonzero.north);
\draw[arr, rounded corners=5pt] (zero.south) |- (merge.west);
\draw[arr, rounded corners=5pt] (nonzero.south) |- (merge.east);

% ---- merge feedback routes before processing the next timed-out branch ----
\coordinate (fbcol) at ($(keep.east)+(12mm,0)$);
\coordinate (feedback) at (fbcol |- next.east);
\draw[line width=1pt, rounded corners=5pt]
    (keep.east) -- (keep.east -| fbcol) |- (feedback);
\draw[line width=1pt, rounded corners=5pt]
    (merge.east) -- node[below] {next $Q$} (merge.east -| fbcol) |- (feedback);
\draw[arr] (feedback) -- (next.east);

\end{tikzpicture}
