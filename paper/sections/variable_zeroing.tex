\section{Recursive splitting}
\label{sec:variable-zeroing}

\subsection{Factor branching with variable zeroing}
\label{subsec:zeroing-branching}

% Motivation for splitting
Suppose the solver does not terminate within a reasonable amount of time on a
hitting set produced by Algorithm~\ref{alg:Branching}.

% What is splitting
Then we can further simplify the system by recursive splitting. In each child
branch, we assume a side condition and run Algorithm~\ref{alg:Branching} on the
simplified system. This process continues until the solver terminates or the
computation budget is exhausted.

% Good properties for conditions on the set of conditions
For completeness, the side conditions must cover all possibilities. They
should also simplify the system for the solver input and work well with
Algorithm~\ref{alg:Refine}.

% What is the condition we use, and why?
In this work, we use \emph{variable zeroing}: recursive splitting with the
side conditions $x=0$ and $x\neq0$ for a selected variable $x$. Our experience with the physical systems considered here suggests that
a variable is often always zero or always nonzero.

% Pseudocode for the method with variable zeroing
\begin{algorithm}[htbp]
\small
\caption{Factor branching with variable zeroing}
\label{alg:ZeroingBranching}
\begin{algorithmic}[1]
    \Require A polynomial system $P$ and a timeout $\tau$
    \Function{ZeroingBranching}{$P,\tau$}
        \State $X \gets \Call{Variables}{P}$
        \State \Return $\Call{SolveByZeroing}{P,X,\emptyset,\emptyset,\tau,\emptyset}$
    \EndFunction
    \Function{SolveByZeroing}{$Q,X,Z,N,\tau,C$}
        \State $(S,T) \gets \Call{FactorBranchingWithTimeout}{Q,\tau,C}$
        \State $U \gets \emptyset$ \Comment{Unresolved branches}

        \ForAll{$R \in T$}
            \If{$X=\emptyset$}
                \State $U \gets U\cup\{(R,Z,N)\}$
            \Else
                \State $x \gets \Call{ChooseVariable}{R,X}$
                \State $R_0 \gets \Call{Substitute}{R,x,0}$
                \State $R_1 \gets \Call{UpdateSaturation}{R,N\cup\{x\}}$
                \State $(S_0,U_0,C) \gets \Call{SolveByZeroing}{R_0,X\setminus\{x\},Z\cup\{x\},N,\tau,C}$
                \State $(S_1,U_1,C) \gets \Call{SolveByZeroing}{R_1,X\setminus\{x\},Z,N\cup\{x\},\tau,C}$
                \State $S \gets S\cup S_0\cup S_1$
                \State $U \gets U\cup U_0\cup U_1$
            \EndIf
        \EndFor
        \State \Return $(S,U,C)$
    \EndFunction
\end{algorithmic}
\end{algorithm}

Algorithm~\ref{alg:ZeroingBranching} uses a global set $C$ of learned UNSAT
cores.

% How the constraints are encoded into the system.
Let $N$ be the variables declared nonzero on the current path.
Subsection~\ref{subsec:nonzero-encoding} describes two ways to encode these
conditions as polynomial equations.

Solutions of the nonzero branch are projected onto the original variables, so
the auxiliary variable does not appear in the output. The zero and nonzero
decisions remain part of the solution-set representation.

For comparison, we repeat the experiment twice: once with the fixed side
conditions $x=1$ and $x\neq1$, and once with each split independently drawing
an integer $a$ uniformly from $\{-10,\ldots,10\}$ and using the side conditions
$x=a$ and $x\neq a$. Both runs have a ten-minute wall-clock limit. Neither
alternative finishes within ten minutes, and in both cases the search has
branches that remain unresolved even after exhausting every variable.

These comparisons demonstrate that zero/nonzero splits are substantially more
effective than arbitrary affine splittings on these structured systems.

\subsection{Choice of variable for splitting}
\label{subsec:zeroing-variable-choice}

The choice of variable for splitting affects the efficiency of the algorithm.
If a variable barely occurs in the current hitting set, recursive calls may
repeat the same work. A variable that occurs in many generators is a more
promising choice. Setting it to zero removes all terms containing it at once.
In the nonzero branch, it is appended to the product in Equation~\eqref
{eq:zeroing-saturation}.

For each candidate variable $x$ that is still available on the current branch, define
\begin{align*}
    g_Q(x) &= \#\{q\in Q : x\text{ occurs in }q\},\\
    t_Q(x) &= \#\{(q,m) : q\in Q,\ m\text{ is a term of }q,\ \deg_x(m) > 0\},\\
    e_Q(x) &= \sum_{q\in Q}\ \sum_{\substack{m\text{ a term of }q,\\ \deg_x(m) > 0}} \deg_x(m).
\end{align*}
Its score is the triple
\[
    h_Q(x)=\bigl(g_Q(x),t_Q(x),e_Q(x)\bigr),
\]
and scores are compared lexicographically.

\subsection{Branch contraction}
\label{subsec:zeroing-branch-contraction}

Let $\mathcal H_0$ and $\mathcal H_1$ be the minimal hitting sets for $x=0$ and
$x\neq0$. Suppose, up to exchanging the two children, that
\[
    \mathcal H_0=\emptyset \qquad \text{and} \qquad \mathcal H_1=\{H\}.
\]
The first child is UNSAT. Theorem~\ref{thm:branch-decomposition} and the
variable-zeroing split give
\[
    V_Y(Q;Z,N)=V_Y(H;Z,N\cup\{x\}).
\]
There is no need to run the branch solver on $H$ before choosing the next
splitting variable.

\subsection{Theoretical analysis}
\label{subsec:zeroing-theoretical-analysis}

The correctness argument uses only that the side conditions cover all
possibilities. Let $Y$ be the original variables, let $A$ be the conditions on
the current branch, and let $V_Y(Q;A)$ be the projection onto $Y$ of the
solutions of $Q$ satisfying $A$. For variable zeroing, we write
$V_Y(Q;Z,N)$ for the conditions $z=0$ for $z\in Z$ and $y\neq0$ for $y\in N$.

\begin{theorem}[Correctness of a covering split]
\label{lem:zeroing-split}
Let $D_1,\ldots,D_k$ be side conditions such that every point satisfying $A$
satisfies at least one $D_i$. If $Q_i$ encodes $Q$ together with $D_i$, then
\[
    V_Y(Q;A)=\bigcup_{i=1}^k V_Y(Q_i;A\cup\{D_i\}).
\]
\end{theorem}

\begin{proof}
Every solution on the left satisfies some $D_i$ and therefore belongs to the
corresponding set on the right. Every set on the right imposes all equations
and conditions from the parent branch, so the reverse inclusion also holds.\qed
\end{proof}

For variable zeroing, the conditions $x=0$ and $x\neq0$ cover all values of
$x$. With $Q_0=Q|_{x=0}$ and
$Q_1=\mathrm{UpdateSaturation}(Q,N\cup\{x\})$, the theorem gives
\begin{equation}
\label{eq:zeroing-split}
    V_Y(Q;Z,N)=V_Y(Q_0;Z\cup\{x\},N)
    \cup V_Y(Q_1;Z,N\cup\{x\}).
\end{equation}

If factor branching with a timeout returns solved representations $S$
and timed-out hitting sets $T$, the proof of
Theorem~\ref{thm:branching-correctness} gives
\begin{equation}
\label{eq:branching-timeout-partition}
    V_Y(Q;A)=\operatorname{Sol}(S)
    \cup\bigcup_{R\in T}V_Y(R;A).
\end{equation}

\begin{theorem}[Correctness of recursive splitting]
\label{thm:zeroing-correctness}
Suppose every split uses side conditions that cover all possibilities. If the
recursive procedure returns $(S,U,C)$ on input $P$, where $U$ contains the
unresolved pairs $(R,A_R)$, then
\[
    V(P)=\operatorname{Sol}(S)
    \cup\bigcup_{(R,A_R)\in U}V_Y(R;A_R).
\]
\end{theorem}

\begin{proof}
Equation~\eqref{eq:branching-timeout-partition} accounts for the solved and
timed-out hitting sets at each call. An unresolved hitting set is either added
to $U$ or replaced by child branches. Theorem~\ref{lem:zeroing-split} shows
that any covering family of child conditions preserves its solution set.
Applying this argument recursively proves the result.\qed
\end{proof}

\begin{corollary}[Completeness of variable zeroing]
\label{cor:zeroing-completeness}
If $U=\emptyset$ in the output of Algorithm~\ref{alg:ZeroingBranching}, then
$\operatorname{Sol}(S)=V(P)$.
\end{corollary}

\begin{theorem}[Termination of variable zeroing]
\label{thm:zeroing-termination}
Algorithm~\ref{alg:ZeroingBranching} terminates if each timed solver call
returns after $\tau$.
\end{theorem}

\begin{proof}
Algorithm~\ref{alg:Refine} terminates by Theorem~\ref{thm:refine-termination}
and returns finitely many hitting sets. Each variable-zeroing split removes one variable
from the finite set $X$, so every recursive path is finite. Therefore the whole
recursion tree is finite.\qed
\end{proof}
