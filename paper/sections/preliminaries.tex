\section{Preliminaries and notation}
\label{sec:preliminaries}

\subsection{Polynomial systems, ideals, and varieties}

Let $K$ be a field, let $\overline K$ be its algebraic closure, and write
$R=K[x_1,\ldots,x_n]$. The input is a finite system
$P=\{p_1,\ldots,p_m\}\subseteq R$ of nonzero polynomials. Unless stated
otherwise, a solution is a point in $\overline K^n$. For a finite set
$Q\subseteq R$, its solution set is
\[
  V(Q)=\{a\in\overline K^n:q(a)=0\text{ for every }q\in Q\}.
\]
We call $Q$ satisfiable (SAT) if $V(Q)\neq\emptyset$ and
unsatisfiable (UNSAT) if
$V(Q)=\emptyset$.

We write $\langle Q\rangle$ for the ideal generated by $Q$.

For $X\subseteq\overline K^n$, its vanishing ideal $I(X)$ contains every
polynomial in $R$ that vanishes on all of $X$. The radical of an ideal $J$ is
$\sqrt J=\{g\in R:g^r\in J\text{ for some }r\geq1\}$. Hilbert's Nullstellensatz
gives
\[
  I(V(Q))=\sqrt{\langle Q\rangle}.
\]
Thus $V(Q)=\emptyset$ exactly when
$\langle
Q\rangle=\langle1\rangle$~\cite{cox2015ideals,kreuzer2000computational}.

\subsection{Saturation and decompositions}
\label{subsec:decompositions}

% Saturation
The saturation of an ideal $J$ by $u\in R$~\cite[Ch.~4, \S4]{cox2015ideals} is
\[
  J:u^\infty=\{g\in R:u^mg\in J\text{ for some }m\geq0\}.
\]

% Definition and uniqueness
An ideal $\mathfrak q$ is primary if $gh\in\mathfrak q$ implies
$g\in\mathfrak q$ or $h^m\in\mathfrak q$ for some $m\geq1$, and then
$\sqrt{\mathfrak q}$ is prime. Every ideal $J$ of $R$ has a minimal primary
decomposition~\cite[Ch.~4, \S8]{cox2015ideals}
\[
  J=\mathfrak q_1\cap\cdots\cap\mathfrak q_t
\]
into primary ideals with distinct radicals
$\mathfrak p_j=\sqrt{\mathfrak q_j}$, where no $\mathfrak q_j$ contains the
intersection of the others. The radicals are the
associated primes of $J$, and the minimal ones among them are the minimal
primes of $J$. The minimal primes intersect to
$\sqrt J$~\cite[Ch.~4, \S6]{cox2015ideals}. The components whose radicals
are minimal primes, called isolated, are unique; the other components, called
embedded, are not~\cite[Ch.~4, \S8]{cox2015ideals}. Saturation keeps the
components whose radicals do not contain $u$,
$J:u^\infty=\bigcap_{u\notin\mathfrak p_j}\mathfrak
q_j$~\cite[Prop.~4.9]{atiyah1969introduction}.
A prime that contains the intersection of finitely many ideals contains one
of them~\cite[Prop.~1.11]{atiyah1969introduction}.

\subsection{Monomial orders and Gr\"obner bases}
\label{subsec:groebner}

Let $\mathbb M_n=\{x^\alpha:\alpha\in\mathbb N^n\}$ be the monomials of $R$.
A monomial order $\prec$ is a total order on $\mathbb M_n$ such that every
nonempty subset has a least element and
\[
  u\prec v \quad\Longrightarrow\quad uw\prec vw
  \qquad (u,v,w\in\mathbb M_n).
\]
For a nonzero polynomial $p$, the leading monomial $\operatorname{LM}(p)$ is
the greatest monomial of $p$ under $\prec$, $\operatorname{LC}(p)$ is its
coefficient, and $\operatorname{LT}(p)$ is its leading term.

For an ideal $J$, define its leading-monomial ideal by
\[
  \operatorname{LM}(J) =\langle\operatorname{LM}(f):f\in J\setminus\{0\}\rangle.
\]
A finite set $G\subseteq J$ is a Gr\"obner basis of $J$ if
\[
  \operatorname{LM}(J) =\langle\operatorname{LM}(g):g\in G\rangle.
\]
A finite set $Q$ is \emph{interreduced} if every element is monic and no term
of one element is divisible by the leading monomial of another.
Algorithm~\ref{alg:interreduction} interreduces a finite set of nonzero
polynomials.

\begin{algorithm}[H]
  \caption{Interreduction}
  \label{alg:interreduction}
  \begin{algorithmic}[1]
    \Require A finite set $Q\subseteq R\setminus\{0\}$ and a monomial
    order $\prec$
    \Ensure An interreduced set that generates $\langle Q\rangle$
    \While{there exist distinct $p,q\in Q$ and a term $cu$ of $p$
    with $\operatorname{LM}(q)\mid u$}
    \State $r\gets p-\dfrac{cu}{\operatorname{LT}(q)}q$
    \State $Q\gets Q\setminus\{p\}$
    \If{$r\neq0$}
    \State $Q\gets Q\cup\{r\}$
    \EndIf
    \EndWhile
    \State $Q\gets\{p/\operatorname{LC}(p):p\in Q\}$
    \State \Return $Q$
  \end{algorithmic}
\end{algorithm}

The loop terminates. Let $\mu(Q)$ be the multiset of the monomials of all
elements of $Q$. An iteration replaces the monomials of $p$ by those of $r$,
which lack $u$ and are monomials of $p$ or strictly smaller than $u$. If $r=0$
or $r$ already lies in $Q$, the iteration only removes the monomials of $p$.
Either way, $\mu(Q)$ decreases in the multiset extension of $\prec$, the
order generated by replacing an element of a multiset with finitely many
smaller ones. It is well founded because $\prec$
is~\cite{dershowitz1979proving}. Interreduction preserves
$\langle Q\rangle$ and $V(Q)$~\cite[Ch.~2, \S7]{cox2015ideals}. Since
$\operatorname{LM}(r)\preceq\operatorname{LM}(p)$, it does not increase the
multiset of leading monomials, and it does not shrink the ideal
$M(Q)=\langle\operatorname{LM}(q):q\in Q\rangle$: an iteration with
$u\neq\operatorname{LM}(p)$ keeps $\operatorname{LM}(p)$ as the leading
monomial of $r$, and if $u=\operatorname{LM}(p)$, then $\operatorname{LM}(q)$
divides it and $q$ stays in $Q$.

If $Q$ is interreduced and a monic $g$ divides some $q\in Q$ with
$\operatorname{LM}(g)\in M(Q)$, then $g=q$. Indeed, some $q'\in Q$ has
$\operatorname{LM}(q')\mid\operatorname{LM}(g)\mid\operatorname{LM}(q)$, so
$q'=q$, and $q/g$ is a constant, equal to $1$ because $g$ and $q$ are monic.

For a fixed monomial order, every ideal $J$ has a unique interreduced
Gr\"obner basis, its reduced Gr\"obner basis.
The reduced basis is $\{1\}$ if and only if $J=\langle1\rangle$. Therefore, a
Gr\"obner basis computation can describe a branch and detect when it is
UNSAT~\cite{cox2015ideals,kreuzer2000computational}.

% Containment test by reduction, used to prune every method's output
Reduction also tests containment. A polynomial $g$ reduces to zero modulo a
finite set $G$ if repeatedly replacing $g$ by
$g-\frac{cu}{\operatorname{LT}(q)}q$, for a term $cu$ of $g$ and $q\in G$
with $\operatorname{LM}(q)\mid u$, reaches $0$. If every element of a finite
set $G'$ reduces to zero modulo $G$, then
$\langle G'\rangle\subseteq\langle G\rangle$, and hence
$V(G)\subseteq V(G')$. If $G$ is a Gr\"obner basis, the test is exact:
$\langle G'\rangle\subseteq\langle G\rangle$ if and only if every element of
$G'$ reduces to zero modulo $G$~\cite[Ch.~2, \S\S3, 6]{cox2015ideals}.

\subsection{Factor sets and hitting sets}

Fix a factorization over $K$ of each polynomial $q$ of a finite system
$Q\subseteq R\setminus\{0\}$:
\[
  q=c_q\prod_{f\in C_q}f^{e_{q,f}},
\]
where $c_q\in K\setminus\{0\}$, the factor set $C_q$ contains the distinct
monic, nonconstant irreducible factors of $q$, and $e_{q,f}\geq1$. We write
$C(Q)=\bigcup_{q\in Q}C_q$.
A point satisfies $q=0$ exactly
when at least one factor in $C_q$ vanishes at that point.

A hitting set of $Q$ is a set $H\subseteq C(Q)$ such that $H\cap
C_q\neq\emptyset$ for every $q\in Q$.
It is minimal if no proper subset is also a hitting
set~\cite{berge1989hypergraphs}.
We write $\mathcal H(Q)$ for the family of minimal hitting sets of $Q$.
Each $H\in\mathcal H(Q)$ defines a branch in which all factors in $H$
vanish.
