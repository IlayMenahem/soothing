\section{Factor branching framework}
\label{sec:method}

\subsection{Framework description}
\label{subsec:branching-framework}

% motivation for breaking down a system of polynomial equations
Polynomial-ideal computations can be intractable even for small systems~\cite{mayr1982complexity}. Thus, decomposing a system into smaller and simpler
subproblems may make the individual solver calls easier.

% How breaking down is done
We break a system into smaller subproblems by alternating two operations:
interreducing the current subproblem, and factoring each generator to
enumerate the minimal hitting sets of the factor sets. Then again do the same
on each hitting set, until the subproblems can't be further broken down.

Here is the pseudocode for this refinement of a polynomial system $Q$ into a smaller set of subproblems.

\begin{algorithm}[htbp]
\caption{Refinement}
\label{alg:Refine}
\begin{algorithmic}[1]
    \Require A finite polynomial system $Q$ and a collection $A$ of normalized systems visited on the current recursion path

    \State $Q\gets Q\setminus\{0\}$
    \State $\widetilde Q\gets\Call{Interreduce}{Q}$ \Comment{Optional; otherwise $\widetilde Q\gets Q$}
    \State Factor each generator of $\widetilde Q$
    \State $\mathcal{H} \gets$ the minimal hitting sets of the factor sets
    \State $T \gets \emptyset$
    \ForAll{$H\in\mathcal H$}
        \If{$H=\widetilde Q$ or $H\in A$}
            \State $T\gets T\cup\{H\}$
        \Else
            \State $T\gets T\cup\Call{Refine}{H,A\cup\{\widetilde Q\}}$
        \EndIf
    \EndFor
    \State \Return $T$

\end{algorithmic}
\end{algorithm}

% How branching is done
Let $\mathcal L(P)$ denote the terminal hitting sets obtained from $P$.
For each $H\in\mathcal L(P)$, the branch solver must either describe $V(H)$ exactly or certify that $V(H)=\emptyset$.

\begin{algorithm}[htbp]
\caption{Factor branching}
\label{alg:Branching}
\begin{algorithmic}[1]
    \Require A set of polynomials $P = \{p_1, \ldots, p_m\}$ and an exact branch solver $\mathcal{B}$.

    \State $S \gets \emptyset$ \Comment{Solution-set representations}
    \State $R \gets \Call{Refine}{P,\emptyset}$

    \ForAll{$H\in R$}
        \State Use $\mathcal{B}$ to solve $H$
        \If{$\mathcal{B}$ returns a representation of $V(H)$}
            \State Add the returned representation to $S$
        \EndIf
    \EndFor

    \State \Return $S$

\end{algorithmic}
\end{algorithm}

The order of the terminal hitting sets affects running time but not
correctness. Optionally, we can sort the hitting sets by their estimated cost to
solve. We use the number of generators in a hitting set as a proxy for the
cost, processing smaller hitting sets first.

\subsection{UNSAT-core learning}
\label{subsec:unsat-core-learning}

We also evaluated UNSAT-core learning, following ideas from polynomial
UNSAT-core extraction and conflict-driven
learning~\cite{sun2016,marques1999grasp,marques2009cdcl}.
After certifying a branch $H$ as UNSAT, this extension extracts a subset
$U\subseteq H$ with $V(U)=\emptyset$ by iterative refinement as in
Section 5 of~\cite{sun2016}. Any later branch containing $U$ can be
discarded, since every solution of that branch would also have to solve $U$.
However, UNSAT-core learning provided no performance benefit on the systems
we tested. We therefore omit it from the algorithms presented here.

% Theoretical analysis of the method
\subsection{Theoretical analysis}
\label{subsec:branching-theoretical-analysis}

Let $S$ be a collection of solution-set representations, and $\operatorname{Sol}(S)$ be the union of the represented sets.
The theoretical analysis does not require $K$ to be algebraically closed. If a
system has no solution over the algebraic closure, it has no solution over $K$.

For a system $Q$, write $\mathcal H(Q)$ for the minimal hitting sets obtained
in one factorization round.

\begin{theorem}[One-round branch decomposition]
\label{thm:branch-decomposition}
\[
    V(Q) = \bigcup_{H \in \mathcal{H}(Q)} V(H).
\]
\end{theorem}

\begin{proof}
Write $Q=\{q_1,\ldots,q_m\}$ and let $F_i$ be the nonconstant irreducible
factors of $q_i$. Take $x \in V(Q)$. Since $q_i(x)=0$, at least one factor in
$F_i$ vanishes at $x$ for every $i$. The set of all factors that vanish at
$x$ is a hitting set of $F_1,\ldots,F_m$. Because the factor sets are finite,
this set contains a minimal hitting set $H$. Every factor in $H$ vanishes at
$x$, so $x \in V(H)$.

For the reverse inclusion, take $x \in V(H)$ for some
$H \in \mathcal{H}(Q)$. The set $H$ intersects every $F_i$, so each $q_i$ has
a factor that vanishes at $x$. Hence $q_i(x)=0$ for every $i$, and $x \in V(Q)$.\qed
\end{proof}

\begin{lemma}[Recursive refinement]
\label{lem:recursive-refinement}
Suppose refinement of $Q$ terminates with the finite collection $\mathcal L(Q)$. Then
\[
    V(Q)=\bigcup_{H\in\mathcal L(Q)}V(H).
\]
\end{lemma}

\begin{proof}
Consider the finite recursion tree produced by Algorithm~\ref{alg:Refine}.
For every internal node $G$, let $\widetilde G$ be its interreduction (or $G$
itself when interreduction is disabled). Interreduction preserves the generated
ideal, so $V(G)=V(\widetilde G)$. Theorem~\ref{thm:branch-decomposition} then
gives
\[
    V(G)=\bigcup_{H\in\mathcal H(\widetilde G)}V(H),
\]
where the sets in $\mathcal H(\widetilde G)$ are precisely the children of
$G$. A bottom-up induction on the finite tree shows that the variety of each
node equals the union of the varieties of the terminal nodes in its subtree.
Applying this statement at the root $Q$ proves the result.\qed
\end{proof}

\begin{theorem}[Termination of refinement]
\label{thm:refine-termination}
Algorithm~\ref{alg:Refine} terminates on every finite polynomial system.
\end{theorem}

\begin{proof}
If interreduction is disabled, every child consists of irreducible polynomials.
Factoring such a child again returns it unchanged, so the condition
$H=\widetilde Q$ makes it terminal.

Suppose interreduction is enabled. For an interreduced system $Q$, let
\[
    M(Q)=\langle\operatorname{LM}(q):q\in Q\rangle.
\]
For every hitting-set child $H$ and every $q\in Q$, some $h\in H$ divides $q$.
Hence $\operatorname{LM}(h)$ divides $\operatorname{LM}(q)$ and
$M(Q)\subseteq M(H)$.

Suppose equality holds. For any $h\in H$, choose $q\in Q$ having $h$ as a
factor. Equality gives some $q'\in Q$ such that
$\operatorname{LM}(q')\mid\operatorname{LM}(h)\mid\operatorname{LM}(q)$.
Because $Q$ is interreduced, this is possible only for $q'=q$. The two outer
leading monomials are therefore equal, so the remaining factor of $q$ is a
unit. Thus $h$ and $q$ are associates, and their monic normalization gives
$h=q$. Applying this argument to the factor selected for every $q\in Q$ gives
$H=Q$, which is terminal.

It remains to account for the next interreduction. In an elementary reduction,
the old leading monomial either remains, or it is already divisible by the
leading monomial of the reducer. Consequently, interreduction cannot shrink
the ideal generated by the current leading monomials. Thus, for every
nonterminal child $H$ and its interreduction $\widetilde H$,
\[
    M(Q)\subsetneq M(H)\subseteq M(\widetilde H).
\]
Therefore $M$ strictly increases along every recursive path.

The ascending-chain condition for monomial ideals rules out an infinite path.
Every node has finitely many children, so K\"onig's lemma implies that the
refinement tree is finite. Therefore Algorithm~\ref{alg:Refine} terminates.\qed
\end{proof}

The following theorem allows the use of Algorithm~\ref{alg:Branching} without loss of correctness.
\begin{theorem}
\label{thm:branching-correctness}
Algorithm~\ref{alg:Branching} is sound and complete for $P$.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:refine-termination}, refinement terminates with a finite
collection $\mathcal L(P)$. Every interreduction round preserves the current
ideal, and every factorization round preserves the zero set by
Theorem~\ref{thm:branch-decomposition}. Lemma~\ref{lem:recursive-refinement}
therefore gives
\[
    V(P)=\bigcup_{H\in\mathcal L(P)}V(H).
\]
Consider any $H\in\mathcal L(P)$. The exact solver either returns an
exact representation of $V(H)$, which the algorithm adds to $S$, or certifies
that $V(H)=\emptyset$. Empty branches therefore make no contribution, while
every represented branch contributes exactly its variety. Hence
\[
    \operatorname{Sol}(S)
    =\bigcup_{H\in\mathcal L(P)}V(H)
    =V(P).
\]
Every representation added to $S$ is exact, which proves soundness, and the
equality proves completeness.\qed
\end{proof}

Finally, we show that the factor branching algorithm terminates.
\begin{theorem}
\label{thm:branching-termination}
If the exact solver is known to terminate, then Algorithm~\ref{alg:Branching} terminates.
\end{theorem}

\begin{proof}
By Theorem~\ref{thm:refine-termination}, the call $R=\Call{Refine}{P,\emptyset}$ terminates and returns a finite collection.
Therefore Algorithm~\ref{alg:Branching} calls the exact solver a finite number of times, and each call terminates by assumption.
Thus Algorithm~\ref{alg:Branching} terminates.\qed
\end{proof}
