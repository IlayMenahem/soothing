\begin{tikzpicture}[
    font=\small,
    every node/.style={align=center},
    blue box/.style={draw=blue!70!cyan, line width=1pt, rounded corners=6pt,
        fill=blue!50!cyan!4, inner sep=7pt},
    loop box/.style={draw=orange!85!black, line width=1pt, rounded corners=6pt,
        fill=yellow!40!orange!12, inner sep=7pt},
    decision/.style={draw=green!55!black, line width=1pt, diamond, aspect=1.8,
        fill=green!60!teal!8, inner sep=1pt},
    green box/.style={draw=green!55!black, line width=1pt, rounded corners=6pt,
        fill=green!60!teal!4, inner sep=7pt},
    red box/.style={draw=red!70, line width=1pt, rounded corners=6pt,
        fill=red!4, inner sep=7pt},
    return box/.style={draw=violet, line width=1pt, rounded corners=6pt,
        fill=violet!6, inner sep=7pt},
    arr/.style={-{Stealth[length=2.6mm]}, line width=1pt},
]

% ---- initialization and refinement ----
\node[blue box] (init) {%
    \textcolor{blue}{\bfseries Initialize}\\
    $P = \{p_1,\ldots,p_m\}$, $C,S \leftarrow \emptyset$};

\node[blue box, right=12mm of init] (factors) {%
    \textcolor{blue}{\bfseries Refine}\\
    Interreduce and factor to obtain $\mathcal L(P)$};

% ---- loop over terminal hitting sets ----
\node[loop box, below=9mm of factors] (loop) {%
    \textcolor{orange!75!black}{\bfseries Next set}\\
    $H\in\mathcal L(P)$};

% ---- unsat-core decision + prune ----
\node[decision, below=6mm of loop] (learned) {%
    Does $H$ contain\\ a core in $C$?};

\node[green box, right=16mm of learned] (prune) {Skip $H$};

% ---- solve ----
\node[blue box, below=6mm of learned] (solve)
    {\textcolor{blue}{\bfseries Solve $H$}};

% ---- two outcome branches ----
\node[red box] (unsat) at ($(solve.south)+(-27mm,-14mm)$) {%
    \textcolor{red}{\bfseries UNSAT}\\
    Learn UNSAT core $U\subseteq H$};

\node[green box] (sat) at ($(solve.south)+(27mm,-14mm)$) {%
    \textcolor{green!45!black}{\bfseries SAT}\\
    Add solutions to $S$};

% ---- bottom rail ----
\coordinate (raill) at ($(unsat.south)+(0,-4mm)$);
\coordinate (railm) at (raill -| solve.south);
\coordinate (railr) at (raill -| sat.south);

% ---- return ----
\node[return box] (return) at ($(railm)+(0,-10mm)$)
    {\textcolor{violet}{\bfseries Return $(S,C)$}};

% ---- arrows: top pipeline ----
\draw[arr] (init) -- (factors);
\coordinate (entry) at ($(loop.north)+(0,4mm)$);
\draw[arr, rounded corners=5pt]
    (factors.south) |- (entry) -- (loop.north);

% ---- loop body ----
\draw[arr] (loop.south) -- (learned.north);
\draw[arr] (learned.east) -- node[above] {Yes} (prune.west);
\draw[arr] (learned.south) -- node[right] {No} (solve.north);
\draw[arr, rounded corners=5pt] (solve.south) |- (unsat.east);
\draw[arr, rounded corners=5pt] (solve.south) |- (sat.west);

% ---- bottom rail into return ----
\draw[line width=1pt] (unsat.south) -- (raill) -- (railr) -- (sat.south);
\draw[arr] (railm) -- node[right] {done} (return);

% ---- feedback: merge both routes before re-entering the loop ----
\coordinate (fbcol) at ($(prune.east)+(10mm,0)$);
\coordinate (feedback) at (fbcol |- loop.east);
\draw[line width=1pt, rounded corners=5pt]
    (prune.east) -- (prune.east -| fbcol) |- (feedback);
\draw[line width=1pt, rounded corners=5pt]
    (railr) -- node[below] {next $H$} (railr -| fbcol) |- (feedback);
\draw[arr] (feedback) -- (loop.east);

\end{tikzpicture}
